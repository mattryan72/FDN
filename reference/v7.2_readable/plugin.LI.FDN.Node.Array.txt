015  
016  if $sector
017   
018   $temp.Node.Array = = array alloc: size=0
019   $dock.Array = = $sector  ->  get dock array from sector
020   $s = size of array $dock.Array
021   while $s > 0
022    dec $s =
023    $dock = $dock.Array[$s]
024    $sub = = $dock  ->  get subtype
025    $race = $dock  ->  get owner race
026    if $race == Player
027     if find $sub in array: $node.Array
028      skip if find $dock in array: $temp.Node.Array
029       append $dock to array $temp.Node.Array
030     end
031    end
032   end
033   append $temp.Node.Array to array $return.Array
034   gosub Sub.In.Range.Sub:
035   
036  else
037

042     dec $s =
043     $dock = $dock.Array[$s]
044     $sub = = $dock  ->  get subtype
045     if find $sub in array: $node.Array
046      skip if find $dock in array: $return.Array
047       append $dock to array $return.Array
048     end
049    end

079    
080    $sub = = $dock  ->  get subtype
081    if find $sub in array: $node.Array
082     if find $sec in array: $node.sector.Array
083      if not find $dock in array: $Node.In.Range.Array

089      end

091    else

093      if find $sub in array: $relay.Array

095        append $dock to array $Relay.In.Range.Array
096      else
097       skip if find $dock in array: $Dock.In.Range.Array

102   end

116   $sort.Array[$cnt] = $jump
117   inc $cnt =
118  end

128  $temp.Array = = array alloc: size=0

147  $temp.Array = = array alloc: size=0
148  $cnt = 0
149  while $cnt < $s
150   $relay = $Relay.In.Range.Array[$cnt]
151   $relay.Sec = $relay  ->  get sector

153   $sort.Array[$cnt] = $jump
