    $sell.stock = $sell.virtual + $sell.physical

  * v9.01 B4/B1: read the trade-task cap fresh here - nothing in this script writes [18]; Sell.Ware (P2-11b) owns it.
  $trade.active = $al.Settings[18]
  $trade.max = $al.Settings[19]
  * v9.01 B4: cap check - if either value is not an int, or max <= 0, treat as unlimited ($cap.ok stays TRUE)
  $cap.ok = [TRUE]
  if is datatyp[ $trade.active ] == DATATYPE_INT
   if is datatyp[ $trade.max ] == DATATYPE_INT
    if $trade.max > 0
     if not $trade.active < $trade.max
      $cap.ok = [FALSE]
     end
    end
   end
  end

  if not $cap.ok
   * v9.01 D2: debug - one line when the cap blocks a sell (dock id keeps it traceable)
   if $debug.on
    $txt.log = $dock + ' Sell skipped - trade task cap reached (' + $trade.active + '/' + $trade.max + ')'

Surplus.Effective.Sub:
 $surplus.raw = $ware.settings[7]
 $surplus.mode = $al.Settings[28]
 * mode: 0 off, 2 v8 compatible, anything else (1, null, other values) counts as 1 (on)
 $mode.val = 1
 if is datatyp[ $surplus.mode ] == DATATYPE_INT
  if $surplus.mode == 0
   $mode.val = 0
  end
  if $surplus.mode == 2
   $mode.val = 2
  end
 end
 * raw kind: 0 = default (-1 or any other value), 1 = on, 2 = off, 3 = null
 $raw.kind = 0
 if is datatyp[ $surplus.raw ] == DATATYPE_NULL
  $raw.kind = 3
 end
 if is datatyp[ $surplus.raw ] == DATATYPE_INT
  if $surplus.raw == 1
   $raw.kind = 1
  end
  if $surplus.raw == 0
   $raw.kind = 2
  end
 end
 $surplus.flag = [FALSE]
 * explicit per-ware on
 if $raw.kind == 1
  $surplus.flag = [TRUE]
 end
 * -1 or other value: follows the default (on for mode 1 and 2)
 if $raw.kind == 0
  if $mode.val == 1
   $surplus.flag = [TRUE]
  end
  if $mode.val == 2
   $surplus.flag = [TRUE]
  end
 end
 * null: on only for mode 1 (v8: null meant the player switched it off)
 if $raw.kind == 3
  if $mode.val == 1
   $surplus.flag = [TRUE]
  end
 end
