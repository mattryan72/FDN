if not $flag
 if not $dock == $dc

  $s = size of array $ware.array
  * v9.01 P8: counter for a short wait every 10 wares
  $ware.count = 0

  if $debug.on
   $txt.log = $dock + ' Pass, ' + $s + ' wares'

  end
  while $s > 0
   dec $s =
   * v9.01 B4: skip flag for this ware - the loop body never breaks out early, so the pass walks to completion and the every-10-wares wait below still runs
   $skip.ware = [FALSE]
   $ware = $ware.array[$s]
   
   * check that the ware transport class is not station
   $transport.class = = get transport class of ware $ware
   * -----------------------------------------------------------------------------------
   
   if not $transport.class == ST
    * this calls the sell sub (cap check + per-dock/per-ware lock inside)
    gosub Sell.Ware.Sub:
    * -----------------------------------------------------------------------------------
    
    * calculate the stock levels
    * v9.01 D3/B4: record via Ware.Rec, create=[FALSE]; null = no record yet - skip this ware (keeps loop walking)
    $ware.settings = [THIS] -> call script lib.LI.FDN.Ware.Rec : station=$dock ware=$ware create=[FALSE]
    if not $ware.settings
     * v9.01 B4: nothing to sell or return for a ware with no record
     $skip.ware = [TRUE]
    end
    
    if not $skip.ware

     $virtual.stock = $ware.settings[0]
     $physical.stock = $dock  ->  get amount of ware $ware in cargo bay
     $stock = $virtual.stock + $physical.stock
     * -----------------------------------------------------------------------------------
     
     * get maintain amount
     $maintain.level = $ware.settings[2]
     if not $maintain.level > 0
      $maintain.level = 0
     end
     * calculate required
     $required.at.dock = $maintain.level - $stock
     if not $required.at.dock > 0
      $surplus.flag = $ware.settings[7]
      if $surplus.flag
       $surplus = $stock - $maintain.level
       if $surplus > 0
        * v9.01 B3: return only while the DC is still valid (no wait between the pair, so this is atomic)
        $dc.valid = [THIS] -> call script lib.LI.FDN.Is.Valid : obj=$dc need.dock=[TRUE]
        if $dc.valid
         = null -> call script plugin.LI.FDN.Update.Ware : ware=$ware  amount=$surplus  station=$dc  action='add'
         = null -> call script plugin.LI.FDN.Update.Ware : ware=$ware  amount=$surplus  station=$dock  action='remove'
         * v9.01 D2: debug (dock id keeps it traceable)
         if $debug.on
          $txt.log = $dock + ' Surplus ' + $surplus + ' of ' + $ware + ' to DC'
          = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
         end
        end
       end
      end
     end
     
     * if required greater than 0
     if $required.at.dock > 0

      * v9.01 B3: DC must be valid before we touch it; invalid means skip this ware, nothing else
      $dc.valid = [THIS] -> call script lib.LI.FDN.Is.Valid : obj=$dc need.dock=[TRUE]
      if not $dc.valid
       $skip.ware = [TRUE]
      end
      
      * v9.01 D3: DC record via Ware.Rec, only when the DC is valid; no record yet - create it, then re-read (P2-12a will move this to Ware.Rec create)

       $dc.ware.settings = [THIS] -> call script lib.LI.FDN.Ware.Rec : station=$dc ware=$ware create=[FALSE]

        = null -> call script plugin.LI.FDN.Update.Ware : ware=$ware  amount=null  station=$dc  action='create'

      end

* sell ware sub - called with $ware (and the dock/DC from locals above)
Sell.Ware.Sub:
 * v9.01 B4/B1: record via Ware.Rec; sell only when [5] is an INT >= 0 and stock is above it (v3 logic, unchanged)
 $sell.go = [FALSE]
 $sell.rec = [THIS] -> call script lib.LI.FDN.Ware.Rec : station=$dock ware=$ware create=[FALSE]
 if $sell.rec
  $sell.cap = $sell.rec[5]
  if is datatyp[ $sell.cap ] == DATATYPE_INT
   if $sell.cap >= 0
    $sell.virtual = $sell.rec[0]
    $sell.physical = $dock -> get amount of ware $ware in cargo bay
    $sell.stock = $sell.virtual + $sell.physical
    if $sell.stock > $sell.cap
     $amount = $sell.stock - $sell.cap
     $sell.go = [TRUE]
    end
   end
  end
 end
 if $sell.go
  * v9.01 B4/B1: read the trade-task cap fresh here - nothing in this script writes [18]; Sell.Ware (P2-11b) owns it.
  $trade.active = $al.Settings[18]
  $trade.max = $al.Settings[19]
  * v9.01 B4: cap check - if either value is not an int, or max <= 0, treat as unlimited ($cap.ok stays TRUE)
  $cap.ok = [TRUE]
  if is datatyp[ $trade.active ] == DATATYPE_INT
   if is datatyp[ $trade.max ] == DATATYPE_INT
    if $trade.max > 0
     if not $trade.active < $trade.max
      $cap.ok = [FALSE]
     end
    end
   end
  end
  * v9.01 B2: ONE block only - cap first, then the per-dock/per-ware lock.
  if not $cap.ok
   * v9.01 D2: debug - one line when the cap blocks a sell (dock id keeps it traceable)
   if $debug.on
    $txt.log = $dock + ' Sell skipped - trade task cap reached (' + $trade.active + '/' + $trade.max + ')'
    = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
   end
  end

  if $cap.ok
   * v9.01 B2: per-dock/per-ware lock - key FDN.Trade.<ware>, held by Sell.Ware (P2-11b) on every exit, stale after 120 s.
   * Lock builds its local from text 4100 as FDN.Busy.FDN.Trade.<ware>; the key string matches REFERENCE.
   $trade.key = 'FDN.Trade.' + $ware
   $lock = [THIS] -> call script lib.LI.FDN.Lock : obj=$dock key=$trade.key action='try' timeout=120
   if $lock
    * v9.01 D2: debug - one line when a sell starts (no tasks-now count; Sell.Ware owns the counter)
    if $debug.on
     $txt.log = $dock + ' Sell cap ' + $sell.cap + ', stock ' + $sell.stock + ', selling ' + $amount + ' of ' + $ware
     = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
    end
    START = null -> call script plugin.LI.FDN.Sell.Ware : dock=$dock ware=$ware amount=$amount
   end
   if not $lock
    * v9.01 D2: debug - one line when the lock blocks a sell (dock id keeps it traceable)
    if $debug.on
     $txt.log = $dock + ' Sell skipped - trade guard held for ' + $ware
     = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
    end
   end
  end
 end
