* Script: plugin.LI.FDN.Supply.Start  (v2.1, engine 50)
* Desc: FDN v9.01 - Starts supply process
* Author: Logain Abler
* -----------------------------------------------------------------------------------
* This script is called by the AL Plugin timer event and starts the supply cycle
* 
* Expected values:
* - N/A the Factory and dock arrays are stored in the AL Plugin variable
* 
* Returns values:
* - N/A
* 
* To Do - P2-09a switches Supply.Factory.Check to read cursor [13] / budget [22] from $al.Settings
* 
* Version: 2.1
* Date: 26/10/2026
* Tested: TC 26/10/2026
* -----------------------------------------------------------------------------------
* v9.01 B1/B3/P1 changes (Supply.Start):
*   B1  [12] is now a real run-lock storing 'playing time' (stale after [22]*4 s), set on
*       the array, not a local copy. [13] becomes the factory cursor (P4 round-robin).
*   B3  The DC is validated every tick via lib.LI.FDN.Is.Valid; if invalid it is cleared to
*       null, the player is told once, and FDN pauses.
*   P1  plugin.LI.FDN.Scan now runs at most every [17] s (default 300) using timestamp [16].
* -----------------------------------------------------------------------------------
* v9.01 Phase 2 changes (P2-08):
*   D1  Header switched to lib.LI.FDN.Cfg.Get (the only allowed copy of the settings fetch).
*   D2  Debug.Sub deleted; every debug line goes through lib.LI.FDN.Log with $s.name as caller.
*   B1  Run-lock on [12] is now acquired/released via lib.LI.FDN.Lock (try/release); stale after
*       [22]*4 s so a crashed Factory.Check cannot wedge the supply loop forever.
*   P4  Factory pass runs synchronously (no START) - Supply.Factory.Check owns its own loop and
*       will read cursor [13] / budget [22] from $al.Settings in P2-09a.
* -----------------------------------------------------------------------------------
* v9.01 Phase 2 changes (P2-14) - debug logging only, no behaviour change:
*   One 'DC: <dc>' line is written per Scan run, after the DC has been validated.
* -----------------------------------------------------------------------------------
$script = = get script name
$s.name = $script
* v9.01 D1: standard header - lib.LI.FDN.Cfg.Get is the only allowed settings fetch (P2-01)
$al.Settings = [THIS] -> call script lib.LI.FDN.Cfg.Get :
if not $al.Settings
 return null
end
$PageID = $al.Settings[1]

* v9.01 P4: read factory budget [22]; fall back to 40 on null/unset (T1: null comparisons are FALSE)
$factory.budget = $al.Settings[22]
if not is datatyp[ $factory.budget ] == DATATYPE_INT
 * missing or wrong type - use the default and persist it so later ticks read a real int
 $factory.budget = 40
 $al.Settings[22] = $factory.budget
 set global variable: name='al.LI.FDN.event' value=$al.Settings
end
if not $factory.budget > 0
 * zero or negative - clamp to the default and persist it
 $factory.budget = 40
 $al.Settings[22] = $factory.budget
 set global variable: name='al.LI.FDN.event' value=$al.Settings
end

* v9.01 P4: read factory cursor [13]; fall back to 0 on null/unset (T1)
$cursor = $al.Settings[13]
if not is datatyp[ $cursor ] == DATATYPE_INT
 $al.Settings[13] = 0
 set global variable: name='al.LI.FDN.event' value=$al.Settings
end
if $cursor < 0
 * negative cursor - wrap to the start (T1: a null would have been caught above)
 $al.Settings[13] = 0
 set global variable: name='al.LI.FDN.event' value=$al.Settings
end

* v9.01 P1: read scan interval [17]; fall back to 300 s on null/unset (T1)
$scan.interval = $al.Settings[17]
if not is datatyp[ $scan.interval ] == DATATYPE_INT
 $scan.interval = 300
end
if not $scan.interval > 0
 $scan.interval = 300
end

* v9.01 P1: read the last scan time [16] (null means never scanned) and grab now once
$scan.time = $al.Settings[16]
$now = = playing time

* -----------------------------------------------------------------------------------

* v9.01 P1: refresh the station arrays at most every [17] s (was every call in v8)
* a null/non-int [16] means 'never scanned' so we run once; Scan stamps [16] itself (P2-06)
$do.scan = [TRUE]
if is datatyp[ $scan.time ] == DATATYPE_INT
 * already scanned before - only re-scan when the interval has passed (T1: null < int is FALSE)
 $elapsed = $now - $scan.time
 if $elapsed < $scan.interval
  $do.scan = [FALSE]
 end
end
if $do.scan
 * v9.01 P1/P4: Scan takes no arguments and stamps [16] itself; the search=4 arg is gone
 START  = [THIS] -> call script plugin.LI.FDN.Scan :
end
 = wait 1000 ms

* v9.01: re-read settings after the wait - Scan may have rewritten [2]/[3]/[4]/[8]/[15]/[16]
* (arrays are shared references, but the values we copy here must reflect any changes Scan made)
$dc = $al.Settings[2]
$factory.array = $al.Settings[3]
$dock.array = $al.Settings[4]
$call.count.dock = $al.Settings[11]

* v9.01: guard the dock counter (T1 - null < 6 is FALSE, so an unset [11] would skip forever)
if not is datatyp[ $call.count.dock ] == DATATYPE_INT
 $call.count.dock = 0
end
if $call.count.dock < 0
 $call.count.dock = 0
end

* v9.01 D2: log every tick via lib.LI.FDN.Log (replaces the old gosub Debug.Sub)
$debug.on = $al.Settings[7]
if $debug.on
 = [THIS] -> call script lib.LI.FDN.Log : text=' Start Supply Called' caller=$s.name
end

* v9.01 B3: no DC set at all - quietly do nothing (no message, no log) so the player is
* only told once, when a real DC first fails validation below
if not $dc
 return null
end

* v9.01 B3: validate the DC via lib.LI.FDN.Is.Valid (replaces the three manual checks;
* returns TRUE / FALSE / null-for-bad-input - any non-TRUE is treated as invalid)
$dc.valid = [THIS] -> call script lib.LI.FDN.Is.Valid : obj=$dc need.dock=[TRUE]

* v9.01 B1: one place to clear the DC, tell the player and pause (see sub below)
if not $dc.valid
 gosub DC.Invalid:
 return null
end

* v9.01 P2-14: one 'DC: <dc>' line per Scan run (not every tick); $do.scan is still set from above
if $do.scan
 if $debug.on
  $txt.log = 'DC: ' + $dc
  = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
 end
end

* v9.01 B3: from here on $dc is a valid, existing, player owned Dock
if $factory.array
 * -----------------------------------------------------------------------------------
 * v9.01 B1/P4 - run-lock via lib.LI.FDN.Lock; stale after [22]*4 s (REFERENCE A row 12)
 * so a crashed Factory.Check cannot wedge the supply loop forever.
 $lock.timeout = $factory.budget * 4
 $locked = [THIS] -> call script lib.LI.FDN.Lock : obj=null key=12 action='try' timeout=$lock.timeout
 if not $locked
  * still held by another tick (younger than the timeout) - skip only the factory pass
  if $debug.on
   = [THIS] -> call script lib.LI.FDN.Log : text='Factory Check already running or recently started - skip' caller=$s.name
  end
 else
  * free (or stale) - take it, run Factory.Check synchronously in this task, release.
  if $debug.on
   = [THIS] -> call script lib.LI.FDN.Log : text='Factory Array found' caller=$s.name
  end
  * v9.01 P4: synchronous call (no START) - Supply.Factory.Check owns its own loop and
  * will read cursor [13] / budget [22] from $al.Settings in P2-09a; we only hold the lock.
  = [THIS] -> call script plugin.LI.FDN.Supply.Factory.Check :
  * release even if Factory.Check exited early - Lock returns TRUE on a clean release,
  * FALSE/null would mean something odd happened (logged by its own guards)
  = [THIS] -> call script lib.LI.FDN.Lock : obj=null key=12 action='release' timeout=null
 end
 * -----------------------------------------------------------------------------------
end

* v9.01: dock pass - runs every 7th tick via the [11] counter (unchanged from P1-01)
if $call.count.dock < 6
 inc $call.count.dock =
 $al.Settings[11] = $call.count.dock
 set global variable: name='al.LI.FDN.event' value=$al.Settings
 return null
else
 $al.Settings[11] = 0
 set global variable: name='al.LI.FDN.event' value=$al.Settings

 if $dock.array
  * v9.01 D2: log via lib.Log (replaces gosub Debug.Sub)
  if $debug.on
   = [THIS] -> call script lib.LI.FDN.Log : text='Dock Array found' caller=$s.name
  end
  * start the dock supply script (P2-10a will switch this to a synchronous call + Lock on its own index)
  START  = null -> call script plugin.LI.FDN.Supply.Dock.Check :
  * -----------------------------------------------------------------------------------
 end
end
return null
* -----------------------------------------------------------------------------------

* v9.01 B3 - clear an invalid DC, tell the player once and pause FDN
* (the reason text is fixed here; lib.LI.FDN.Is.Valid no longer fills $txt.debug)
DC.Invalid:
$al.Settings[2] = null
set global variable: name='al.LI.FDN.event' value=$al.Settings
 * v9.01 B3: tell the player once (logbook entry; same command on AP and FL)
write to player logbook 'FDN: Distribution Centre lost or invalid - FDN paused. Pick a new DC in the FDN menu.'
if $debug.on
 = [THIS] -> call script lib.LI.FDN.Log : text='DC invalid - cleared and paused' caller=$s.name
end
endsub
* -----------------------------------------------------------------------------------
return null
