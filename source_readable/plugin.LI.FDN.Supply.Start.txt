* Script: plugin.LI.FDN.Supply.Start  (v2.0, engine 50)
* Desc: FDN v9.01 - Starts supply process
* Author: Logain Abler
* -----------------------------------------------------------------------------------
* This script ic called by the AL Plugin timer event and starts the supply cycle
* 
* Expected values:
* - N/A the Factory and dock arrays are stored in the AL Plugin variable
* 
* Returns values:
* - N/A
* 
* To Do - N/A
* 
* Version: 2.0
* Date: 17/10/2026
* Tested: not tested
* -----------------------------------------------------------------------------------
* v9.01 B1/B3/P1 changes (Supply.Start):
*   B1  [12] is now a real run-lock storing 'playing time' (stale after 120 s), set on
*       the array, not a local copy. [13] becomes the reverse toggle (handled by P1-02).
*   B3  The DC is validated every tick (exists / owner race == Player / Dock class);
*       if invalid it is cleared to null, the player is told once, and FDN pauses.
*   P1  Sector.Array now runs at most every 300 s using timestamp [16].
* -----------------------------------------------------------------------------------
* - DC
* - factory arrray - holds all the know player factories
* - dock arrray - holds all the know player docks
* Note: The two arrays are updated each timethe main menu opens, but it is the
* supply logic which removes any destroyed stations
$script = = get script name
$al.Settings = get global variable: name='al.LI.FDN.event'
if $al.Settings
 $PageID = $al.Settings[1]
 $DebugID = 0 + $PageID
 $dc = $al.Settings[2]
 $factory.array = $al.Settings[3]
 $dock.array = $al.Settings[4]
 $call.count.dock = $al.Settings[11]
 * v9.01 B1: [12] is the run-lock (playing time of last start; null = free)
 $is.running = $al.Settings[12]
 * v9.01 P1: [16] holds the playing time of the last Sector.Array scan
 $scan.time = $al.Settings[16]
 * v9.01 B1/B3/P1: current playing time, used for the lock and scan stamps
 $now = = playing time
else
 return null
end

* -----------------------------------------------------------------------------------

* v9.01 P1: refresh the station arrays at most every 300 s (was every call)
* a null [16] means 'never scanned' so we run once and stamp it
$do.scan = [TRUE]
if $scan.time
 * already scanned before - only re-scan when 300 s have passed
 $elapsed = $now - $scan.time
 if $elapsed < 300
  $do.scan = [FALSE]
 end
end
if $do.scan
 START  = [THIS] -> call script plugin.LI.FDN.Sector.Array : search=4
 $al.Settings[16] = $now
 set global variable: name='al.LI.FDN.event' value=$al.Settings
end
 = wait 1000 ms

if $al.Settings[7]
 $txt.debug = ' Start Supply Called'
 gosub Debug.Sub:
end

* v9.01 B3: no DC set at all - quietly do nothing (no message, no log) so the player is
* only told once, when a real DC first fails validation below
if not $dc
 return null
end

* v9.01 B3: validate the DC - it must exist, be player owned and a Dock
* (v8 only checked 'if $dc', which misses destroyed / sold / captured stations)
$dc.valid = [TRUE]
if not $dc -> exists
 * destroyed or never set
 $dc.valid = [FALSE]
 $txt.debug = 'DC does not exist'
end
if $dc.valid
 $owner = $dc -> get owner race
 if not $owner == Player
  * sold, captured or given away
  $dc.valid = [FALSE]
  $txt.debug = 'DC is not owned by the player'
 end
end
if $dc.valid
 if not $dc -> is of class Dock 2007
  * wrong station type for a DC
  $dc.valid = [FALSE]
  $txt.debug = 'DC is not a Dock class station'
 end
end

* v9.01 B3: one place to clear the DC, tell the player and pause (see sub below)
if not $dc.valid
 gosub DC.Invalid:
 return null
end

* v9.01 B3: from here on $dc is a valid, existing, player owned Dock
if $factory.array
 * debug
 * -----------------------------------------------------------------------------------
 
 * v9.01 B1 - real run-lock on [12] (was a local flag that never engaged the guard)
 * [12] holds the playing time of the last Factory.Check start; null = free, stale after 120 s
 $skip.factory = [FALSE]
 if $is.running
  * locked - only let this tick through when the lock is more than 120 s old (stale)
  $elapsed = $now - $is.running
  if $elapsed < 120
   $skip.factory = [TRUE]
   if $al.Settings[7]
    $txt.debug = 'Factory Check already running or recently started - skip'
    gosub Debug.Sub:
   end
  end
 end
 if not $skip.factory
  * free (or stale) - take the lock on the array, then start Factory.Check
  $al.Settings[12] = $now
  set global variable: name='al.LI.FDN.event' value=$al.Settings
  * note: this script never releases [12]; plugin.LI.FDN.Supply.Factory.Check does (P1-02)
  if $al.Settings[7]
   $txt.debug = 'Factory Array found'
   gosub Debug.Sub:
  end
  START  = null -> call script plugin.LI.FDN.Supply.Factory.Check :
 end
 * -----------------------------------------------------------------------------------
end

* this check skips the dock supply until call 3 times
if $call.count.dock < 6
 inc $call.count.dock =
 $al.Settings[11] = $call.count.dock
 return null
else
 $al.Settings[11] = 0

 if $dock.array
  * debug
  if $al.Settings[7]
   $txt.debug = 'Dock Array found'
   gosub Debug.Sub:
  end
  * start the dock supply script
  START  = null -> call script plugin.LI.FDN.Supply.Dock.Check :
  * -----------------------------------------------------------------------------------
 end
end
return null
* -----------------------------------------------------------------------------------

* v9.01 B3 - clear an invalid DC, tell the player once and pause FDN
* $txt.debug holds the reason (set by the caller before the gosub)
DC.Invalid:
$al.Settings[2] = null
set global variable: name='al.LI.FDN.event' value=$al.Settings
 * v9.01 B3: tell the player once (logbook entry; same command on AP and FL)
write to player logbook 'FDN: Distribution Centre lost or invalid - FDN paused. Pick a new DC in the FDN menu.'
if $al.Settings[7]
 $txt.debug = $txt.debug + ' - cleared and paused'
 gosub Debug.Sub:
end
endsub
* -----------------------------------------------------------------------------------

* Debug Massage handling
Debug.Sub:
$d.time = = playing time
$d.time = = format time: $d.time
$txt = $d.time + ';' + $script + ';' + $txt.debug
write to log file $DebugID  append=[TRUE]  value=$txt
endsub
* -----------------------------------------------------------------------------------
return null
