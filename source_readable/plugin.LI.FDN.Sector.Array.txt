* Script: plugin.LI.FDN.Sector.Array  (v3, engine 50)
* Desc: FDN v9.01 - Thin wrapper: returns the cached sector list from lib.LI.FDN.Sectors
* Arg 0: search (Number) 'search'
* Author: Logain Abler
* -----------------------------------------------------------------------------------
* The station scan now lives in plugin.LI.FDN.Scan (run by Supply.Start) and the cache
* reader in lib.LI.FDN.Sectors. This script keeps its name and argument for the menus
* that still call it and returns the cached sector list (sorted, reversed) or null.
*
* Expected values:
* - 1 = factory sectors, 2 = dock sectors, 3 = all trade sectors (adds advanced satellite sectors)
* - anything else (the menus pass 4) = factory sectors plus dock sectors, as in v8
*
* Returns values:
* - an array of sectors, or null if there are none or the cache is not built yet
*   (if the cache was never built, Scan is started once so the next call has data)
*
* To Do - P3 switches the menu callers to lib.LI.FDN.Sectors and this wrapper goes away
*
* Version: 3.0
* Date: 26/10/2026
* Tested: TC 26/10/2026
* -----------------------------------------------------------------------------------
 * v9.01 B10/D1 changes (Sector.Array):
 *   B10 the station scan and the list return were one script; this is now a wrapper over the cache.
 *   D1  header via lib.LI.FDN.Cfg.Get.
 * -----------------------------------------------------------------------------------

$al.Settings = [THIS] -> call script lib.LI.FDN.Cfg.Get :
if not $al.Settings
 return null
end
$s.name = = get script name
$debug.on = $al.Settings[7]

* only 1, 2 and 3 are real modes; the menus pass 4 and v8 treated every other value as factories plus docks
$mode = $search
if not is datatyp[ $mode ] == DATATYPE_INT
 $mode = 4
end
if $mode < 1
 $mode = 4
end
if $mode > 4
 $mode = 4
end

$result = null
if $mode == 4
 $result = = array alloc: size=0
 $from.array = [THIS] -> call script lib.LI.FDN.Sectors : mode=1
 gosub Append.Sub:
 $from.array = [THIS] -> call script lib.LI.FDN.Sectors : mode=2
 gosub Append.Sub:
 $result = = sort array $result
 $result = reverse array $result
else
 $result = [THIS] -> call script lib.LI.FDN.Sectors : mode=$mode
end

$count = 0
if is datatyp[ $result ] == DATATYPE_ARRAY
 $count = size of array $result
end
if $count > 0
 return $result
end

* nothing to return: if the cache was never built (no scan stamp) start Scan once so the next call has data
$scan.time = $al.Settings[16]
if not is datatyp[ $scan.time ] == DATATYPE_INT
 if $debug.on
  $txt.log = 'Sector.Array - sector cache not built yet, starting Scan'
  = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
 end
 START  = [THIS] -> call script plugin.LI.FDN.Scan :
end
return null

Append.Sub:
* append every sector of $from.array that is not in $result yet (a null or empty list adds nothing)
if is datatyp[ $from.array ] == DATATYPE_ARRAY
 $n = size of array $from.array
 while $n > 0
  dec $n =
  $sector = $from.array[$n]
  if not find $sector in array: $result
   append $sector to array $result
  end
 end
end
endsub
