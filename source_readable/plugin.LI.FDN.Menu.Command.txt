* Script: plugin.LI.FDN.Menu.Command (v1, engine 50)
* Desc: FDN v9.01 - Menu command handler: runs one menu return value and returns the new view stack
* Arg 0: stack (Value) 'stack'
* Arg 1: return (Value) 'return'

* Author: Logain Abler
* -----------------------------------------------------------------------------------
* Phase 3 command handler for the FDN menu. Main.Menu builds one view from the top of its
* local view stack, opens it, then passes the menu return value to this script. The script
* returns the NEW view stack, or null to close the menu. Main.Menu only assigns the result
* back, it never edits the stack itself. There are no FDND.* globals and no 'continue'.

* The stack is an array of entries [view, obj, ware]. View numbers are the v8 values of
* settings [5]: 1 = DC, 2 = Stations, 4 = Factory.Details, 5 = Dock, 6 = Ware.Details,
* 7 = Config.Menu (3 is retired).

* The return value is either an array [command, value, ware, extra], the string of a
* global menu action, or -1 (Escape, handled like 'back').

* Arguments:
* - stack (Value): the current view stack (array of [view, obj, ware])
* - return (Value): the menu return value described above

* Returns values:
* - the new view stack: after navigation commands, a successful remove.ware and master.reset
* - the same stack, unchanged: after edit commands and global actions
* - null: after 'back' or Escape on a stack with a single entry (close the menu)

* To Do - N/A

* Version: 1.0
* Date: 27/10/2026
* Tested: TC 27/10/2026
* -----------------------------------------------------------------------------------

$s.name = = get script name
$al.Settings = [THIS] -> call script lib.LI.FDN.Cfg.Get :
if not $al.Settings
 return null
end
$PageID = $al.Settings[1]
if not is datatyp[ $stack ] == DATATYPE_ARRAY
 return null
end

* split the return value into command, value, ware and extra (missing parts stay null)
$command = null
$value = null
$ware = null
$extra = null
if is datatyp[ $return ] == DATATYPE_ARRAY
 $rs = size of array $return
 if $rs > 0
  $command = $return[0]
 end
 if $rs > 1
  $value = $return[1]
 end
 if $rs > 2
  $ware = $return[2]
 end
 if $rs > 3
  $extra = $return[3]
 end
else
 $command = $return
 if is datatyp[ $return ] == DATATYPE_INT
  if $return == -1
   $command = 'back'
  end
 end
end

$s = size of array $stack
if $s < 1
 return null
end

* --- Navigation ------------------------------------------------------------------
if $command == 'back'
 if $s < 2
  return null
 end
 $n = $s - 1
 $new.stack = array alloc: size=$n
 $i = 0
 while $i < $n
  $new.stack[$i] = $stack[$i]
  $i = $i + 1
 end
 return $new.stack
end

$view = 0
$obj = $value
if $command == 'open.dc'
 $view = 1
 $obj = $al.Settings[2]
end
if $command == 'open.stations'
 $view = 2
end
if $command == 'open.factory'
 $view = 4
end
if $command == 'open.dock'
 $view = 5
end
if $command == 'open.ware'
 $view = 6
end
if $command == 'open.config'
 $view = 7
 $obj = null
 $ware = null
end
if $view > 0
 $entry = array alloc: size=3
 $entry[0] = $view
 $entry[1] = $obj
 $entry[2] = $ware
 append $entry to array $stack
 return $stack
end

end

if $command == 'set.dc'
 gosub Set.DC.Sub:
 return $new.stack
end

* Unknown command: just leave the stack untouched --------------------------------------
return $stack


Ware.Prep.Sub:
* validate the station, read or create the ware record, build the record pointer
$ok = 1
if is datatyp[ $value ] == DATATYPE_STATION
 $station = $value
else
 $station = $al.Settings[2]
end
$valid = [THIS] -> call script lib.LI.FDN.Is.Valid : obj=$station need.dock=[FALSE]
if not $valid == [TRUE]
 $ok = 0
end
if $ok > 0
 $rec = [THIS] -> call script lib.LI.FDN.Ware.Rec : station=$station ware=$ware create=[TRUE]
 if not $rec
  $ok = 0
 end
end
$pointer = = sprintf: pageid=$PageID textid=4000, $ware, null, null, null, null
endsub

Stock.Handler.Sub:
* v8 Regroom.Ware.Sub: to.virtual moves the physical stock into the virtual count, to.physical moves virtual stock into the cargo bay as far as it fits
gosub Ware.Prep.Sub:
if $ok == 0
 endsub
end
$virtual = $rec[0]
if not $virtual
 $virtual = 0
end
if $command == 'to.virtual'
 $amount = $station -> get amount of ware $ware in cargo bay
 $virtual = $virtual + $amount
 $remove = 0 - $amount
  = $station -> add $remove units of $ware
else
 $amount = $station -> get free amount of ware $ware in cargo bay
 if $virtual <= $amount
   = $station -> add $virtual units of $ware
  $virtual = 0
 else
   = $station -> add $amount units of $ware
  $virtual = $virtual - $amount
 end
end
$rec[0] = $virtual
gosub Ware.Save.Sub:
endsub

Global.Factory.Sub:
* v8 Global.Factory.Sub: switch product [0] or resource [1] on or off for every factory in settings [3]
$fa = $al.Settings[3]
if not is datatyp[ $fa ] == DATATYPE_ARRAY
 endsub
end
$pointer = = read text: page=$PageID id=4001
$k = size of array $fa
while $k > 0
 $k = $k - 1
 $factory = $fa[$k]
 if is datatyp[ $factory ] == DATATYPE_STATION
  $exists = $factory -> exists
  if $exists
   $fset = $factory -> get local variable: name=$pointer
   if is datatyp[ $fset ] == DATATYPE_ARRAY
    if $gval == 1
     $fset[$gidx] = 1
    else
     $fset[$gidx] = null
    end
    $factory -> set local variable: name=$pointer value=$fset
   end
  end
 end
end

Debug.Cycle.Sub:
* B5: settings [7] cycles -1 -> 1 -> 0 -> -1, null counts as -1
$cur = $al.Settings[7]
if is datatyp[ $cur ] == DATATYPE_NULL
 $cur = -1
end
if $cur == -1
 $al.Settings[7] = 1
else if $cur == 1
 $al.Settings[7] = 0
else
 $al.Settings[7] = -1
end

 = [THIS] -> call script lib.LI.FDN.Log : text=$log.txt caller=$s.name
set global variable: name='al.LI.FDN.event' value=$al.Settings
endsub

Set.DC.Sub:
* validate the dock, ask one confirmation only when a DC is already set, let DC.Change do the switch (it owns the settings [2] write), then restart the stack on the DC view
$new.stack = $stack
$dock = $value
$valid = [THIS] -> call script lib.LI.FDN.Is.Valid : obj=$dock need.dock=[TRUE]
if not $valid == [TRUE]
 endsub
end
$old.dc = $al.Settings[2]
if $old.dc
 $name.txt = [THIS] -> call script plugin.LI.FDN.Format.Name : object=$dock
 $txt = = sprintf: pageid=$PageID textid=351, $name.txt, null, null, null, null
 $confirm = null -> get user input: type=Var/Boolean, title=$txt
 if not $confirm == [TRUE]
  endsub
 end
end
$res = [THIS] -> call script plugin.LI.FDN.DC.Change : old=$old.dc new=$dock
if $res == 'busy'
 $txt = = read text: page=$PageID id=352
 write to player logbook $txt
 endsub
end
* any other refusal (invalid, same, fl.managed) leaves the stack as it was
if not $res == 'ok'
 endsub
end
 = [THIS] -> call script lib.LI.FDN.Reg.Add : station=$dock
$log.txt = 'FDN Menu: set.dc done'
 = [THIS] -> call script lib.LI.FDN.Log : text=$log.txt caller=$s.name
$entry = array alloc: size=3
$entry[0] = 1
$entry[1] = $dock
$new.stack = array alloc: size=1
$new.stack[0] = $entry
endsub
