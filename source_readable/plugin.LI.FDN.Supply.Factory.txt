* Script: plugin.LI.FDN.Supply.Factory  (v1.0, engine 50)
* Desc: FDN v8 - Supply Factory logic
* Arg 0: factory (Value) 'factory'
* Arg 1: flag (Value) 'flag'

001  * Author: Logain Abler
002  * -----------------------------------------------------------------------------------
003  * This script is called by LI.FDN.Supply.Factory.Check
004  * It is used to handle supply between the factrry and DC
005  
006  * Expected values:
007  * - Factory
008  * - Flag - 1 for Product 2 for Resources
009  
010  * Returns values:
011  * - N/A
012  
013  * To Do - N/A
014  
015  * Version: 1.0
016  * Date: 22/05/2012
017  * Tested: 22/05/2012
018  * -----------------------------------------------------------------------------------
019  
020  * Fetch values saved in the AL Plugin global variable:
021  * - DC
022  $script = = get script name
023  $al.Settings = get global variable: name='al.LI.FDN.event'
024  if $al.Settings
025   $PageID = $al.Settings[1]
026   $DebugID = 0 + $PageID
027   $dc = $al.Settings[2]
028  else
029   return null
030  end
031  * -----------------------------------------------------------------------------------
032  
033  * flag 1 is for products
034  if $flag == 1
035   $ware = $factory  ->  get product ware
036   gosub Add.Product.Sub:
037   return null
038   
039  else
040   $ware.array = $factory  ->  get tradeable ware array from station
041   $s = size of array $ware.array
042   while $s > 0
043    dec $s =
044    $ware = $ware.array[$s]
045    if $factory  ->  uses ware $ware as primary resource
046     gosub Supply.Resource.Sub:
047    end
048   end
049  end
050  return null
051  * -----------------------------------------------------------------------------------
052  
053  Add.Product.Sub:
054  * This sub is called to pass products to the DC
055  * For resources the wares will be added to the DC in the following order:
056  * - Physical Storage
057  * - Virtual Storage
058  
059  * Get the product amount to be passed to the DC
060  $product.amount = $factory  ->  get amount of ware $ware in cargo bay
061  * debug
062  if $al.Settings[7]
063   $txt.debug = $factory + ' has ' + $product.amount + ' of ' + $ware + ' to pass to DC'
064   gosub Debug.Sub:
065  end
066  * add to dc
067   = null -> call script plugin.LI.FDN.Update.Ware : ware=$ware  amount=$product.amount  station=$dc  action='add'
068  * -----------------------------------------------------------------------------------
069  
070  * Remove ware from factory
071  $remove = - $product.amount
072   = $factory  ->  add $remove units of $ware
073  * debug
074  if $al.Settings[7]
075   $txt.debug = 'Remove ' + $product.amount + '  of ' + $ware + ' from ' + $factory
076   gosub Debug.Sub:
077  end
078  endsub
079  * -----------------------------------------------------------------------------------
080  
081  Supply.Resource.Sub:
082  * This sub is called to pass resources to the factory
083  * For resources the wares will be removed from the DC in the following order:
084  * - Virtual Storage
085  * - Physical Storage
086  * - Buy if not in storage
087  
088  * calculate what th factory needs
089  $required = = $factory  ->  get number of resources per cycle for ware $ware
090  $current = $factory  ->  get amount of ware $ware in cargo bay
091  $required = ( $required * 3 ) - $current
092  $max = $factory  ->  get max amount of ware $ware that can be stored in cargo bay
093  if $max < $required
094   $required = $max
095  end
096  * -----------------------------------------------------------------------------------
097  
098  * debug
099  if $al.Settings[7]
100   $txt.debug = $factory + ' requires ' + $required + '  of ' + $ware
101   gosub Debug.Sub:
102  end
103  
104  * if statement to ensure no negative values
105  if $required > 0
106   * remove the required from the DC and return what the DC was able to supply
107   $from.dc = null -> call script plugin.LI.FDN.Update.Ware : ware=$ware  amount=$required  station=$dc  action='remove'
108   * -----------------------------------------------------------------------------------
109   
110   
111   * check if the DC was able to supply the full amount if not call the buy script
112   if $from.dc == $required
113     = $factory  ->  add $required units of $ware
114   else
115    if $from.dc > 0
116      = $factory  ->  add $from.dc units of $ware
117    end
118    * calculate what needs to be bought
119    $required.buy = $required - $from.dc
120    
121    * debug
122    if $al.Settings[7]
123     $txt.debug = 'Buy ' + $required.buy + ' of ware ' + $ware + ' for ' + $factory
124     gosub Debug.Sub:
125    end
126    
127    * call the buy script
128    $pointer = = sprintf: pageid=$PageID textid=4000, $ware, null, null, null, null
129    $ware.settings = $dc  ->  get local variable: name=$pointer
130    * if can buy
131    if $ware.settings[3]
132     START  = null -> call script plugin.LI.FDN.Buy.Ware : factory=$factory  ware=$ware  amount=$required.buy
133    end
134    
135   end
136  end
137  endsub
138  * -----------------------------------------------------------------------------------
139  
140  * Debug Massage handling
141  Debug.Sub:
142  $d.time = = playing time
143  $d.time = = format time: $d.time
144  $txt = $d.time + ';' + $script + ';' + $txt.debug
145  write to log file $DebugID  append=[TRUE]  value=$txt
146  endsub
147  * -----------------------------------------------------------------------------------
148  return null
