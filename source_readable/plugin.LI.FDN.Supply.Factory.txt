* v9.01 B3/B11 changes (Supply.Factory):
*   B3  The DC is validated once near the top of this task (exists / owner race == Player
*       / Dock class) into $dc.valid. If invalid, products are not taken from the factory
*       and no resource remove or buy is attempted (P1-01 already pauses FDN and tells
*       the player; no message here). The product removal only takes what Update.Ware
*       reports as added - only a null return falls back to v8 behaviour (0 included).
*       Tighten after P1-04: once 'add' returns real amounts on every path, remove the null
*       fallback so a partial add can never leave wares in both stations.
*   B3  The DC ware record is checked for null before [3] is read; if missing, only that
*       ware's buy is skipped (flag), the remaining wares are still processed.
*   B11 Buying a shortfall STARTs plugin.LI.FDN.Buy.Ware at most once per 120 s per factory:
*       FDN.Settings [3] holds the playing time of the last buy start; while
*       'playing time' - [3] < 120 no further Buy.Ware is STARTed for this factory.
* -----------------------------------------------------------------------------------

* Fetch values saved in the AL Plugin global variable:
* - DC
$script = = get script name
$al.Settings = get global variable: name='al.LI.FDN.event'
if $al.Settings
 $PageID = $al.Settings[1]
 $DebugID = 0 + $PageID
 $dc = $al.Settings[2]
 * v9.01 B3/B11 - names and time, read once for the whole task
 $pointer.settings = = read text: page=$PageID id=4001
 $now = = playing time
else
 return null
end
* -----------------------------------------------------------------------------------

* v9.01 B3: validate the DC once per task (a task runs atomically between waits, so this
* stays true for both subs). P1-01 clears and pauses on an invalid DC every tick, so in
* practice $dc.valid is TRUE here - this check stops the unchecked uses of $dc in case a
* DC dies mid-tick. No message: Supply.Start already told the player once.
$dc.valid = [TRUE]
if not $dc -> exists
 * destroyed or never set
 $dc.valid = [FALSE]
end
if $dc.valid
 $owner = $dc -> get owner race
 if not $owner == Player
  * sold, captured or given away
  $dc.valid = [FALSE]
 end
end
if $dc.valid
 if not $dc -> is of class Dock 2007
  * wrong station type for a DC
  $dc.valid = [FALSE]
 end
end
* -----------------------------------------------------------------------------------

* flag 1 is for products
if $flag == 1
 $ware = $factory  ->  get product ware
 gosub Add.Product.Sub:
 return null
 
else
 $ware.array = $factory  ->  get tradeable ware array from station
 $s = size of array $ware.array
 while $s > 0
  dec $s =
  $ware = $ware.array[$s]
  if $factory  ->  uses ware $ware as primary resource
   gosub Supply.Resource.Sub:
  end
 end
end
return null

Add.Product.Sub:
* This sub is called to pass products to the DC
* For resources the wares will be added to the DC in the following order:
* - Physical Storage
* - Virtual Storage

* Get the product amount to be passed to the DC
$product.amount = $factory  ->  get amount of ware $ware in cargo bay
* debug
if $al.Settings[7]
 $txt.debug = $factory + ' has ' + $product.amount + ' of ' + $ware + ' to pass to DC'
 gosub Debug.Sub:
end
* v9.01 B3: only touch the product if the DC is still valid (exists / player owned / Dock)
if not $dc.valid
 * debug
 if $al.Settings[7]
  $txt.debug = 'DC invalid - product kept at ' + $factory
  gosub Debug.Sub:
 end
 return null
end
* add to dc and take the amount Update.Ware says it accepted
$added.to.dc = null -> call script plugin.LI.FDN.Update.Ware : ware=$ware  amount=$product.amount  station=$dc  action='add'
* -----------------------------------------------------------------------------------

* v9.01 B3: remove from the factory only what was added to the DC (0 included) - only a null
* return falls back to v8 behaviour (e.g. Update.Ware before P1-04). Tighten after P1-04:
* once 'add' returns real amounts on every path, drop the null fallback so a partial add can
* never leave wares in both stations.
$remove = - $product.amount
if not is datatyp[ $added.to.dc ] == DATATYPE_NULL
 $remove = - $added.to.dc
end
* Remove ware from factory
 = $factory  ->  add $remove units of $ware
* debug
if $al.Settings[7]
 * v9.01: negate on its own line - a unary minus inside a longer expression is mis-compiled
 $removed = - $remove
 $txt.debug = 'Remove ' + $removed + '  of ' + $ware + ' from ' + $factory
 gosub Debug.Sub:
end
endsub

Supply.Resource.Sub:
* This sub is called to pass resources to the factory
* For resources the wares will be removed from the DC in the following order:
* - Virtual Storage
* - Physical Storage
* - Buy if not in storage

* v9.01 B3: skip flag for this ware only (the sub is called per ware, so a fresh flag
* here never carries over) and it keeps the loop walking when we bail out below
$skip.ware = [FALSE]
if not $dc.valid
 * v9.01 B3: no resource remove or buy while the DC is invalid
 $skip.ware = [TRUE]
end

* calculate what th factory needs
$required = = $factory  ->  get number of resources per cycle for ware $ware
$current = $factory  ->  get amount of ware $ware in cargo bay
$required = ( $required * 3 ) - $current
$max = $factory  ->  get max amount of ware $ware that can be stored in cargo bay
if $max < $required
 $required = $max
end

* debug
if $al.Settings[7]
 if not $skip.ware
  $txt.debug = $factory + ' requires ' + $required + '  of ' + $ware
  gosub Debug.Sub:
 end
end

* if statement to ensure no negative values
if $required > 0
 * v9.01 B3: the DC ware record is read only when the DC is valid and exists; if it is
 * missing (e.g. a freshly set DC with no FDN stock yet) skip this ware's buy and keep on
 * with the remaining wares - do not return from inside the sub here, that would end the
 * whole task and skip them
 $have.record = [FALSE]
 if $dc.valid
  $pointer = = sprintf: pageid=$PageID textid=4000, $ware, null, null, null, null
  $ware.settings = $dc  ->  get local variable: name=$pointer
  if $ware.settings
   $have.record = [TRUE]
  end
 end

 * remove the required from the DC and return what the DC was able to supply
 if not $skip.ware
  $from.dc = null -> call script plugin.LI.FDN.Update.Ware : ware=$ware  amount=$required  station=$dc  action='remove'
  * -----------------------------------------------------------------------------------
  
  

  * check if the DC was able to supply the full amount if not call the buy script
  if $from.dc == $required
   = $factory  ->  add $required units of $ware
  else
   if $from.dc > 0
    = $factory  ->  add $from.dc units of $ware
   end
   * calculate what needs to be bought
   $required.buy = $required - $from.dc
   
   * debug
   if $al.Settings[7]
    $txt.debug = 'Buy ' + $required.buy + ' of ware ' + $ware + ' for ' + $factory
    gosub Debug.Sub:
   end
   
   * v9.01 B3: only look at the record when it exists (null would make [3] unreadable)
   if $have.record
    * if can buy
    if $ware.settings[3]
     * v9.01 B11: skip starting Buy.Ware while a previous buy for this factory is still in
     * flight - FDN.Settings [3] holds the playing time of its start, stale after 120 s
     * (same pattern as the P1-01 run-lock on al.LI.FDN.event [12])
     $factory.settings = $factory  ->  get local variable: name=$pointer.settings
     if not $factory.settings
      * should not happen - Factory.Check allocates it; create it so a buy can still start
      $settings.new = = array alloc: size=6
      $factory  ->  set local variable: name=$pointer.settings value=$settings.new
      $factory.settings = $factory  ->  get local variable: name=$pointer.settings
     end
     $busy.until = $factory.settings[3]
     $busy = [FALSE]
      * v9.01 B11: only an integer is a start time (FDN v7 saves kept a station in [3]) - VERIFY
     if is datatyp[ $busy.until ] == DATATYPE_INT
      * v9.01 B11: never write a combined comparison - compute the elapsed time first
      $elapsed = $now - $busy.until
      if $elapsed < 120
       $busy = [TRUE]
       * debug
       if $al.Settings[7]
        $txt.debug = 'Buy skipped - factory busy for another ' + ( 120 - $elapsed ) + ' s'
        gosub Debug.Sub:
       end
      end
     end
     if not $busy
      * v9.01 B11: mark this factory busy from now on (start time; stale after 120 s)
      $factory.settings[3] = $now
      $factory  ->  set local variable: name=$pointer.settings value=$factory.settings
      START  = null -> call script plugin.LI.FDN.Buy.Ware : factory=$factory  ware=$ware  amount=$required.buy
     end
    end
   end
   
  end
 end
end
endsub
