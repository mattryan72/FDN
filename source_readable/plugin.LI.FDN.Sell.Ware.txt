* Script: plugin.LI.FDN.Sell.Ware  (v3, engine 50)
* Desc: FDNv9 - Sell ware from a dock to NPC stations in the trade sectors
* Arg 0: dock (Value) 'dock'
* Arg 1: ware (Value) 'ware'
* Arg 2: amount (Value) 'amount'

* Author: Logain Abler
* -----------------------------------------------------------------------------------
* This script is STARTed by plugin.LI.FDN.Supply.Dock when a dock holds stock above its
* sell cap for a ware. It walks every cached trade sector, finds NPC buyers at or above
* the sell-at price and sells what it can through lib.LI.FDN.Trade (home = the dock, whose
* account receives; no dest). Supply.Dock holds the per-dock/per-ware trade guard
* FDN.Trade.<ware> while this runs - this script owns releasing it at Exit:. The task
* counter [18] is owned by this script (incremented here, decremented at Exit:); it does
* NOT take the trade guard itself because its caller already holds it.

* Expected values:
* - Dock    station to sell from (player owned)
* - Ware    ware to sell
* - Amount  hint only - units above the cap when Supply.Dock looked; recomputed each pass

* Returns values:
* - N/A (null)

* To Do - N/A

* Version: 3.0
* Date: 26/10/2026
* Tested: TC 26/10/2026
* -----------------------------------------------------------------------------------
* v9.01 P2-11b changes - closes D1, D2 (one of nine), B4, B12, B15; owns counter [18]:
*   D1  Standard header via lib.LI.FDN.Cfg.Get ($PageID is no longer needed here).
*   D2  Debug.Sub and Check.Alive.Sub deleted; every debug line goes through
*       lib.LI.FDN.Log with $s.name as caller.
*   B4  The dock ware record is read via Ware.Rec(create=[FALSE]) - the sell cap [5] must be
*       an INT >= 0 (-1 = selling off) and the sell-at price [6] must be an INT, both up front
*       and on every loop pass (the record is re-read each pass).
*   B12 Dock and DC are validated with lib.LI.FDN.Is.Valid up front and again after
*       every wait in the sector loop; any failure ends the pass at Exit:.
*   B15 The temp-drone block, the Update.Ware 'remove' call and all manual stock removal
*       are replaced by one wait-free lib.LI.FDN.Trade call per found buyer (hardened drone,
*       money/stock/space clamps; sell mode already removes the sold units from dock stock
*       and stamps rec[8] - calling Update.Ware here would remove them twice).
*   B2/B18 Trade ownership: Supply.Dock takes FDN.Trade.<ware> on the dock (try, stale 120 s)
*       BEFORE it STARTs this script and does not release it, so Sell.Ware only RELEASES that
*       lock at Exit:. It owns [18]: increments while below [19], decrements at Exit: with a
*       floor at 0. The single Exit: label balances both on every route.
*   Sector search unchanged from v8 - cached trade sectors (mode 3), one find per pass, wait
*   500-1000 ms between passes with a re-validation after the wait.

* -----------------------------------------------------------------------------------
$counter.inc = [FALSE]

* v9.01 B2/B18: the trade guard key is built before any guard that can exit, so Exit:
* always knows whether to release it (only when the ware is not null).
if $ware
 $trade.key = 'FDN.Trade.' + $ware
end

$s.name = = get script name
$al.Settings = [THIS] -> call script lib.LI.FDN.Cfg.Get :
if not $al.Settings
 goto label Exit:
end
$dc = $al.Settings[2]
$debug.on = $al.Settings[7]
$exclude.array = $al.Settings[8]

* v9.01 B12: the dock must be a valid player dock before anything runs
$dock.valid = [THIS] -> call script lib.LI.FDN.Is.Valid : obj=$dock need.dock=[TRUE]
if not $dock.valid
 if $debug.on
  $txt.log = $dock + ' Sell skipped - dock no longer valid'
  = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
 end
 goto label Exit:
end

* v9.01 B12: the DC must be a valid player dock before anything runs
$dc.valid = [THIS] -> call script lib.LI.FDN.Is.Valid : obj=$dc need.dock=[TRUE]
if not $dc.valid
 if $debug.on
  $txt.log = $dock + ' Sell skipped - DC no longer valid'
  = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
 end
 goto label Exit:
end

* v9.01 B4/D3: the dock ware record via Ware.Rec (read-only); sell cap [5] and sell-at
* price [6] must both be INTs before anything runs (Trade re-checks the price live).
$ware.settings = [THIS] -> call script lib.LI.FDN.Ware.Rec : station=$dock ware=$ware create=[FALSE]
if not $ware.settings
 if $debug.on
  $txt.log = $dock + ' Sell skipped - no FDN ware record for ' + $ware
  = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
 end
 goto label Exit:
end
$sell.cap = $ware.settings[5]
if not is datatyp[ $sell.cap ] == DATATYPE_INT
 if $debug.on
  $txt.log = $dock + ' Sell skipped - sell cap not set for ' + $ware
  = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
 end
 goto label Exit:
end
if not $sell.cap >= 0
 if $debug.on
  $txt.log = $dock + ' Sell skipped - sell cap off for ' + $ware
  = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
 end
 goto label Exit:
end
$sell.at = $ware.settings[6]
if not is datatyp[ $sell.at ] == DATATYPE_INT
 if $debug.on
  $txt.log = $dock + ' Sell skipped - sell at price not set for ' + $ware
  = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
 end
 goto label Exit:
end

* v9.01 B2/B18: task cap - increment [18] only while below [19]; a non-int or max <= 0
* means unlimited (same nested INT checks as the Supply.Dock sell gate and Buy.Ware).
$trade.active = $al.Settings[18]
$trade.max = $al.Settings[19]
if is datatyp[ $trade.active ] == DATATYPE_INT
 if is datatyp[ $trade.max ] == DATATYPE_INT
  if $trade.max > 0
   if not $trade.active < $trade.max
    if $debug.on
     $txt.log = $dock + ' Sell skipped - trade task cap reached (' + $trade.active + '/' + $trade.max + ')'
     = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
    end
    goto label Exit:
   end
  end
 end
end
 * [18] is null until Cleanup has run in this save - count from 0 (null + 1 is not safe)
if not is datatyp[ $trade.active ] == DATATYPE_INT
 $trade.active = 0
end

$al.Settings[18] = $trade.active + 1
set global variable: name='al.LI.FDN.event' value=$al.Settings
$counter.inc = [TRUE]

* v9.01 P2-11b: cached trade sectors (mode 3 - factories + docks + advanced satellites);
* the cache never scans, so a null or empty list only means nothing is cached yet.
$sector.array = [THIS] -> call script lib.LI.FDN.Sectors : mode=3
if not $sector.array
 if $debug.on
  $txt.log = $dock + ' Sell skipped - no trade sectors in cache for ' + $ware
  = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
 end
 goto label Exit:
end

$s = size of array $sector.array
$more = [TRUE]
while $s > 0
 dec $s =
 * v9.01 B4/D3: re-read the record each pass; sell cap only when [5] is an INT >= 0
 $ware.settings = [THIS] -> call script lib.LI.FDN.Ware.Rec : station=$dock ware=$ware create=[FALSE]
 if not $ware.settings
  if $debug.on
   $txt.log = $dock + ' Sell stopped - no FDN ware record for ' + $ware
   = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
  end
  goto label Exit:
 end
 $sell.cap = $ware.settings[5]
 if not is datatyp[ $sell.cap ] == DATATYPE_INT
  if $debug.on
   $txt.log = $dock + ' Sell stopped - sell cap not set for ' + $ware
   = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
  end
  goto label Exit:
 end
 if not $sell.cap >= 0
  if $debug.on
   $txt.log = $dock + ' Sell stopped - sell cap off for ' + $ware
   = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
  end
  goto label Exit:
 end
 * v9.01 B4: the amount from Supply.Dock is only a hint - recompute it every pass
 $sector = $sector.array[$s]
 $virtual.stock = $ware.settings[0]
 $physical.stock = $dock  ->  get amount of ware $ware in cargo bay
 $stock = $virtual.stock + $physical.stock
 $amount = $stock - $sell.cap
 if not $amount > 0
  * nothing above the cap - stop looking (no continue, no extra dec)
  $s = 0
  $more = [FALSE]
 end
 if $amount > 0
  * Find stations buying the ware in this sector (v8 find, unchanged)
  $sell.Station = find station: resource $ware with best price: min.price=$sell.at, amount=null, max.jumps=0, startsector=$sector, trader=null, exclude array=$exclude.array
  if $sell.Station
   * v9.01 B15/R-T1: the whole drone dance is inside Trade - hardened spawn, foe/price
   * money/space clamps (T5 confirmed on FL); sell mode removes the sold units from dock
   * stock and stamps rec[8], so no Update.Ware call here (it would remove them twice).
   $traded = [THIS] -> call script lib.LI.FDN.Trade : mode='sell' partner=$sell.Station ware=$ware amount=$amount home=$dock
   if not is datatyp[ $traded ] == DATATYPE_INT
    $traded = 0
   end
   if $traded > 0
    $amount = $amount - $traded
    if $debug.on
     $txt.log = $dock + ' Sold ' + $traded + ' of ' + $ware + ' at ' + $sell.Station
     = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
    end
   end
   if not $amount > 0
    * enough sold - no need to look in the remaining sectors (no continue, no extra dec)
    $s = 0
    $more = [FALSE]
   end
  end
 end

 * v9.01 B12: keep the v8 throttle between passes - one short random wait only when there
 * is more to sell, then re-validate both stations; a failure after the wait ends at Exit:.
 if $more
  = wait randomly from 500 to 1000 ms
  $dock.valid = [THIS] -> call script lib.LI.FDN.Is.Valid : obj=$dock need.dock=[TRUE]
  if not $dock.valid
   if $debug.on
    $txt.log = $dock + ' Sell stopped - dock no longer valid'
    = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
   end
   goto label Exit:
  end
  $dc.valid = [THIS] -> call script lib.LI.FDN.Is.Valid : obj=$dc need.dock=[TRUE]
  if not $dc.valid
   if $debug.on
    $txt.log = $dock + ' Sell stopped - DC no longer valid'
    = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
   end
   goto label Exit:
  end
 end
end
 * v9.01 B15: loop finished with stock still above the cap - no buyer found in any sector
if $amount > 0
 if $debug.on
  $txt.log = $dock + ' No buyer found for remaining ' + $amount + ' of ' + $ware
  = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
 end
end

* -----------------------------------------------------------------------------------

Exit:
 * v9.01 B2/B18: the single exit label - every path ends here, so the trade guard and
 * the task counter are balanced on every route (guards before either is held land here
 * with $counter.inc still [FALSE]). The guard was taken by Supply.Dock on this dock; we
 * only release it. Lock returns null for a destroyed dock, so the station check alone is safe.
if not is datatyp[ $dock ] == DATATYPE_STATION
 $trade.key = null
end

if $ware
 if $trade.key
  = [THIS] -> call script lib.LI.FDN.Lock : obj=$dock key=$trade.key action='release' timeout=null
 end
end
if $counter.inc
 * re-read the settings - another task may have written them while we ran
 $al.Settings = [THIS] -> call script lib.LI.FDN.Cfg.Get :
 if $al.Settings
  $active = $al.Settings[18]
  if is datatyp[ $active ] == DATATYPE_INT
   $active = $active - 1
   if $active < 0
    * floor at zero so a stale or crashed counter can never go negative
    $active = 0
   end
   $al.Settings[18] = $active
   set global variable: name='al.LI.FDN.event' value=$al.Settings
  end
 end
end
return null
