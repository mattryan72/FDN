036  * -----------------------------------------------------------------------------------

043  $temp.array = = array alloc: size=4
044  $temp.array[0] = 1

047  $txt = $value  ->  get sector

049  add custom menu info line to array $menu: text=$temp.array

051  * This call the plugin.LI.FDN.Sort.Wares which returns a wares arrays sorted by type
052  * $sorted.ware.array[x] is the Main Type
053  * $sorted.ware.array[x][x] is the sorted Ware Array for the Main Type
054  $sorted.ware.array = [THIS] -> call script plugin.LI.FDN.Sort.Wares : station=$value

056  $sm = size of array $sorted.ware.array

058  * Create and size show ware array
059  $pointer = = read text: page=$PageID id=4003
060  $show.type.array = $value  ->  get local variable: name=$pointer

066

069  add custom menu heading to array $menu: title=$array
070  * -----------------------------------------------------------------------------------
071  * Start of While Loop with 1 nested While Loop - First loop is Main Type

073  while $sm > 0

075   * -----------------------------------------------------------------------------------
076   * Fetch main.type and ware.array from sorted.ware.array

078   $ware.array = $sorted.ware.array[$sm][1]
079   * -----------------------------------------------------------------------------------
080   $sw = size of array $ware.array
081   if $sw > 0
082    * -----------------------------------------------------------------------------------
083    * if ware.show.array for the main type is null set to 1 (1 = show 2 = hide)
084    if not $show.type.array[$sm]
085     $show.type.array[$sm] = 1
086    end
087    * -----------------------------------------------------------------------------------
088    * -----------------------------------------------------------------------------------
089    * Call the Sub which updates builds the ware header by main type
090    $check = $show.type.array[$sm]
091    gosub Get.Ware.Header.Sub:
092    if $check == 1
093

097      $ware = $ware.array[$sw]
098      * -----------------------------------------------------------------------------------
099      * Calls the global variable for the dynamic menu created by plugin.LI.FDN.Menu.Dock.Details_D

103      append $summary to array $menu
104      * -----------------------------------------------------------------------------------
105     end
106    end

109  end
110  
111  $txt = = read text: page=$PageID id=195

116  $return.array[1] = $value

131  if $check == 1
132   $temp.array = = array alloc: size=10
133   $temp.array[0] = 1

136   $temp.array[1] = $txt

152   $txt = = read text: page=$PageID id=$main.type
153   $txt = = sprintf: pageid=$PageID textid=19, $txt, $sw, null, null, null
154   $temp.array[1] = $txt
155  end
156  
157  $return.array = = array alloc: size=2
158  $return.array[0] = 'show.type.dock'
159  
160  $return.value = = array alloc: size=2
161  $return.value[0] = $sm
162  $return.value[1] = $value
163  
164  $return.array[1] = $return.value

166  $format = = create new array, arguments=$temp.array, $return.array, null, null, null
