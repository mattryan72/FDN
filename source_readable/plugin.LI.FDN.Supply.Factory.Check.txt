* Script: plugin.LI.FDN.Supply.Factory.Check  (v4, engine 50)
* Desc: FDNv9 - Checks factory to be supplied
* Author: Logain Abler
* -----------------------------------------------------------------------------------
* This script is called by LI.FDN.Menu.Supply.Start script if there is a valid
* factory array stored in the AL Plugin variable
* The script contains the checks made on the factory before calling LI.FDN.Menu.Supply.Factory
* The script will also remove any non-existing or non-player-owned factories from the stored array
* NOTE: no else / else if inside the while body - flags and separate ifs are used instead
* -----------------------------------------------------------------------------------
* v9.01 Phase 2 changes (P2-09a):
*   D1  Header switched to lib.LI.FDN.Cfg.Get (the only allowed copy of the settings fetch).
*   D2  Debug.Sub deleted; every debug line goes through lib.LI.FDN.Log with $s.name as caller.
*   B3  Per-factory validity is one call to lib.LI.FDN.Is.Valid : obj=$factory need.dock=[FALSE] -
*       missing, sold, captured or given-away stations are dropped from the stored array [3].
*   D10 Non-managed stations (docks, complex hubs) left in [3] are also removed via
*       lib.LI.FDN.Is.Factory : station=$factory.
*   D4  The per-factory get-or-create and first-seen auto-manage block is replaced by one call to
*       lib.LI.FDN.Fac.Cfg : factory=$factory auto.manage=1 (the pointer text id=4001 read moves into the lib).
*   P4  Supply.Factory now runs synchronously in this task (no START) - one wait-free pass over up to
*       [22] factories per call; after every wait, $enabled=[0] and the DC are re-validated before continuing.
*   P4  Forward round-robin only: starts at cursor [13], processes at most budget [22] valid factories,
*       writes the next cursor back to [13]; removals do not advance the cursor or count against the budget.
*   B1  This script no longer touches run-lock [12] - Supply.Start owns both try and release via
*       lib.LI.FDN.Lock (P2-08 coupling rule).
* -----------------------------------------------------------------------------------
* Expected values:
* - N/A the Factory array is stored in the AL Plugin variable
*
* Returns values:
* - N/A
*
* To Do - N/A
*
* Version: 4.0
* Date: 23/10/2026
* Tested: not tested (M-05 staged 29/10/2026)
 * v9.01 D6 (M-05h): factory loop stops only when Dc.List is empty
* -----------------------------------------------------------------------------------

* v9.01 D1: standard header - lib.LI.FDN.Cfg.Get is the only allowed settings fetch (P2-01)
$script = = get script name
$s.name = $script
$al.Settings = [THIS] -> call script lib.LI.FDN.Cfg.Get :
if not $al.Settings
 return null
end
* Fetch values saved in the AL Plugin global variable:
* - factory array - holds all the know player factories
$factory.array = $al.Settings[3]

* v9.01 B1: nothing to check - Supply.Start owns the run-lock, so just reset the cursor and exit
if not $factory.array
 $al.Settings[13] = 0
 set global variable: name='al.LI.FDN.event' value=$al.Settings
 return null
end
$size.of.array = size of array $factory.array
if not $size.of.array > 0
 $al.Settings[13] = 0
 set global variable: name='al.LI.FDN.event' value=$al.Settings
 return null
end

* v9.01 P4: read factory budget [22]; fall back to 40 on null/unset (T1: null comparisons are FALSE)
$factory.budget = $al.Settings[22]
if not is datatyp[ $factory.budget ] == DATATYPE_INT
 * missing or wrong type - use the default and persist it so later ticks read a real int
 $factory.budget = 40
 $al.Settings[22] = $factory.budget
 set global variable: name='al.LI.FDN.event' value=$al.Settings
end
if not $factory.budget > 0
 * zero or negative - clamp to the default and persist it
 $factory.budget = 40
 $al.Settings[22] = $factory.budget
 set global variable: name='al.LI.FDN.event' value=$al.Settings
end

* v9.01 P4: read factory cursor [13]; out-of-range or invalid cursors start at 0 (T1)
$cursor = $al.Settings[13]
if not is datatyp[ $cursor ] == DATATYPE_INT
 $cursor = 0
end
if $cursor < 0
 $cursor = 0
end
if $cursor >= $size.of.array
 * array shrank since the last pass (removals) - wrap to the start
 $cursor = 0
end
$al.Settings[13] = $cursor
set global variable: name='al.LI.FDN.event' value=$al.Settings
$s = $cursor
$attempted = 0


* copy the debug flag to a local before the loop - array elements are never used directly in conditions (T1)
$debug.on = $al.Settings[7]

* v2 - modify wait depending on number of factories
if $size.of.array > 500
 $wait.1 = 20
 $wait.2 = 50
else if $size.of.array > 300
 $wait.1 = 50
 $wait.2 = 100
else
 $wait.1 = 100
 $wait.2 = 200
end

* v9.01 P4: forward round-robin only - no in-loop wrap, so a budget larger than the array can never
* supply the same factories twice in one call (P2-08 test step 4)
$loop.active = [TRUE]
while $loop.active
 * compute the two loop bounds into a flag before any work (no function calls in while conditions)
 if not $s < $size.of.array
  $loop.active = [FALSE]
 end
 if $attempted >= $factory.budget
  $loop.active = [FALSE]
 end

  $factory = $factory.array[$s]
  $removed = [FALSE]
  * reset the settings local so a removed station can never fall through to Supply.Factory with the
  * previous iteration's array still in it (stale value bug)
  $settings.array = null
  * v9.01 B3: one call covers exists + player ownership (null for missing, FALSE for not owned)
  $valid = [THIS] -> call script lib.LI.FDN.Is.Valid : obj=$factory need.dock=[FALSE]
  if not $valid
   * missing, sold, captured or given away - remove in place; the next entry has moved into slot $s,
   * so do not advance the cursor and do not count this against the budget (P2-08 rule)
   remove element from array $factory.array at index $s
   dec $size.of.array =
   $removed = [TRUE]
   if $debug.on
    = [THIS] -> call script lib.LI.FDN.Log : text=' Clean from Factory Array' caller=$s.name
   end
  end
  * v9.01 D10: a valid station that is not a managed factory (dock, complex hub) never gets supplied
  if $valid
   $is.factory = [THIS] -> call script lib.LI.FDN.Is.Factory : station=$factory
   if not $is.factory
    remove element from array $factory.array at index $s
    dec $size.of.array =
    $removed = [TRUE]
    if $debug.on
     = [THIS] -> call script lib.LI.FDN.Log : text=' Clean from Factory Array' caller=$s.name
    end
   end
  end

  * v9.01 D4: get-or-create the per-factory settings in one lib call; auto.manage=1 applies the
  * global [9] default on first find only, so a player can still change them with auto manage on
  if $valid
   if not $removed
    $settings.array = [THIS] -> call script lib.LI.FDN.Fac.Cfg : factory=$factory auto.manage=1
   end
  end
  if $settings.array
   * copy the two managed flags to locals before any condition (P1 lesson)
   $manage.prod = $settings.array[0]
   $manage.res = $settings.array[1]

    * v9.01 P4: product branch - Supply.Factory now runs synchronously in this task (no START)
    if $manage.prod
     if $debug.on
      = [THIS] -> call script lib.LI.FDN.Log : text=' Supply Factory called flag = 1' caller=$s.name
     end
     * call the supply factory script, flag one for the factory to pass its product to the DC
     = [THIS] -> call script plugin.LI.FDN.Supply.Factory : factory=$factory flag=1
    end

    * v9.01 P4: resource branch - re-check the factory first in case Supply.Factory waited and it
    * was sold or destroyed mid-pass, then run the stalled / near-completion check (v9.01 flags)
    if $manage.res
     * Supply.Factory may have waited - the factory must still exist and be player owned
     $factory.valid = [THIS] -> call script lib.LI.FDN.Is.Valid : obj=$factory need.dock=[FALSE]
     if $factory.valid
      * check.time and full.time is used to see if the factory is stalled
      $check.time = $factory -> get remaining production cycle time
      $full.time = $factory -> get production cycle time: account for secondary resources=[FALSE]
       if $debug.on
        $txt.log = ' Check time = ' + $check.time + ' Full Time = ' + $full.time
        = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
       end
       * v9.01 P4: flags instead of else - stalled, or less than 45 seconds of production left; the
       * log line names which branch fired (matching v3), built in a local so no inline + reaches a call
       $call.resource = [FALSE]
       if $check.time == $full.time
        $call.resource = [TRUE]
        if $debug.on
         $txt.log = ' Production stalled - Supply Resource called'
         = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
        end
       end
       if not $call.resource
        if $check.time < 45
         $call.resource = [TRUE]
         if $debug.on
          $txt.log = ' Production near completion - Supply Resource called'
          = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
         end
        end
       end
       * call the supply factory script to supply resources to the factory (synchronous, P4)
       if $call.resource
        = [THIS] -> call script plugin.LI.FDN.Supply.Factory : factory=$factory flag=null
       end
      end
     end
    end
   * v9.01 P4: only a valid factory that was actually processed counts toward the budget; removals
   * leave the cursor on slot $s so the entry shifted into it is checked next
   if not $removed
    inc $s =
    inc $attempted =
   end
   * wait before the next factory - Supply.Factory waited inside, so this is already spread out
   = wait randomly from $wait.1 to $wait.2 ms
   * v9.01 P4: after every wait, re-check FDN is enabled and the DC still exists / player owned /
   * is a Dock before continuing - Supply.Start owns telling the player when it dies
   $enabled = $al.Settings[0]
   $debug.on = $al.Settings[7]
   if not $enabled
    $loop.active = [FALSE]
   end
   if $loop.active
    * v9.01 D6 (M-05h): stop only when no DC in [30] is valid (Dc.List cleans and returns the list);
    * older settings (< 40) return null -> fall back to the [2] test as before
    $dc.list = [THIS] -> call script lib.LI.FDN.Dc.List :
    if is datatyp[ $dc.list ] == DATATYPE_ARRAY
     $dl.size = size of array $dc.list
     if not $dl.size > 0
      * every DC lost mid-pass - stop the loop; Supply.Start tells the player next tick
      $loop.active = [FALSE]
     end
    else
     $dc = $al.Settings[2]
     $dc.valid = [THIS] -> call script lib.LI.FDN.Is.Valid : obj=$dc need.dock=[TRUE]
     if not $dc.valid
      * DC lost mid-pass - stop the loop; Supply.Start clears it and tells the player next tick
      $loop.active = [FALSE]
     end
    end
   end
  * recompute the loop flag for the next pass of the while body
  if not $s < $size.of.array
   $loop.active = [FALSE]
  end
  if $attempted >= $factory.budget
   $loop.active = [FALSE]
  end
end
* v9.01 P4: write the next cursor back to [13]; wrap at the end of the array so a later
* re-scan that added factories resumes from 0. The run-lock [12] is untouched (B1 - Supply.Start owns it).
if not $s < $size.of.array
 * reached the end of the array (or it emptied out) - start the next pass at 0
 $s = 0
end
$al.Settings[13] = $s
set global variable: name='al.LI.FDN.event' value=$al.Settings

return null
