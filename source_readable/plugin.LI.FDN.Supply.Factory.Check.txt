* Script: plugin.LI.FDN.Supply.Factory.Check  (v3, engine 50)
* Desc: FDNv9 - Checks factory to be supplied
* Author: Logain Abler
* -----------------------------------------------------------------------------------
* This script is called by LI.FDN.Menu.Supply.Start script if there is a valid
* factory array stored in the AL Plugin variable
* The script contains the checks made on the factory before calling LI.FDN.Menu.Supply.Factory
* The script will also remove any non-existing or non-player-owned factories from the stored array
* v9.01 B3:  ownership check added (sold, captured or given-away stations are dropped)
* v9.01 B13: walks the array by index (direction alternates) instead of reverse array,
*            so removals always hit the stored array [3]
* v9.01 B1:  [12] is the run-lock set by Supply.Start; this script releases it on every exit
*            [13] is the direction toggle (new index, [12] was misused for this in v8)
* v9.01 P7:  text 4001 read once before the loop, not inside it
* NOTE: no else / else if inside the while body - flags and separate ifs are used instead
*
* Expected values:
* - N/A the Factory array is stored in the AL Plugin variable
*
* Returns values:
* - N/A
*
* To Do - N/A
*
* Version: 3.0
* Date: 18/10/2026
* Tested: not tested
* -----------------------------------------------------------------------------------

* Fetch values saved in the AL Plugin global variable:
* - factory array - holds all the know player factories
$script = = get script name
$al.Settings = get global variable: name='al.LI.FDN.event'
if $al.Settings
 $PageID = $al.Settings[1]
 $DebugID = 0 + $PageID
 $factory.array = $al.Settings[3]
else
 return null
end
* -----------------------------------------------------------------------------------

* v9.01 B1: nothing to check - release the run-lock before every early return
if not $factory.array
 $al.Settings[12] = null
 set global variable: name='al.LI.FDN.event' value=$al.Settings
 return null
end
$size.of.array = size of array $factory.array
if not $size.of.array > 0
 $al.Settings[12] = null
 set global variable: name='al.LI.FDN.event' value=$al.Settings
 return null
end
* -----------------------------------------------------------------------------------

* v9.01 B13: [13] set = walk backwards this pass and clear it; [13] null = walk forwards and set it
$iterate.backwards = [FALSE]
if $al.Settings[13]
 $iterate.backwards = [TRUE]
 $al.Settings[13] = null
end
if not $iterate.backwards
 $al.Settings[13] = 1
end
set global variable: name='al.LI.FDN.event' value=$al.Settings
$s = 0
if $iterate.backwards
 $s = $size.of.array - 1
end
* -----------------------------------------------------------------------------------

* v2 - modify wait depending on number of factories
if $size.of.array > 500
 $wait.1 = 20
 $wait.2 = 50
else if $size.of.array > 300
 $wait.1 = 50
 $wait.2 = 100
else
 $wait.1 = 100
 $wait.2 = 200
end
* -----------------------------------------------------------------------------------

* v9.01 P7: name of the factory local variable that holds the FDN settings - read once
$pointer = = read text: page=$PageID id=4001
* -----------------------------------------------------------------------------------

* start of While Loop - Loop is for Factories
while [TRUE]
 * v9.01 B13: bound checks - forwards uses the shrinking $size.of.array
 if not $iterate.backwards
  if not $s < $size.of.array
   break
  end
 end
 if $iterate.backwards
  if not $s >= 0
   break
  end
 end
 $factory = $factory.array[$s]
 $removed = [FALSE]
 * v9.01 B3: factory must exist AND be owned by the player
 $valid = [FALSE]
 if $factory -> exists
  $owner = $factory -> get owner race
  if $owner == Player
   $valid = [TRUE]
  end
 end
 if not $valid
  * missing, sold, captured or given away - remove from the stored array
  remove element from array $factory.array at index $s
  dec $size.of.array =
  $removed = [TRUE]
  * debug
  if $al.Settings[7]
   $txt.debug = ' Clean from Factory Array'
   gosub Debug.Sub:
  end
 end
 if $valid
  * fetch local variable that hold the faction settings
  if not $factory -> get local variable: name=$pointer
   $settings.array = array alloc: size=6
   $factory -> set local variable: name=$pointer value=$settings.array
  end
  $settings.array = $factory -> get local variable: name=$pointer
  * check if this is a new factory and flag as found
  * I use this to only manage the setting on first find and not every search
  * so player can still change settings with auto manage on
  if not $settings.array[2]
   $settings.array[2] = 1
   * check auto manage setting if set turn on at factory
   if $al.Settings[9]
    $settings.array[0] = 1
    $settings.array[1] = 1
   end
   $factory -> set local variable: name=$pointer value=$settings.array
  end
  * if product is managed call plugin.LI.FDN.Supply.Factory: factory
  if $settings.array[0]
   * debug
   if $al.Settings[7]
    $txt.debug = ' Supply Factort called flag = 1'
    gosub Debug.Sub:
   end
   * call the supply factory script, flag one for the factory to pass its product to the DC
   START null -> call script plugin.LI.FDN.Supply.Factory : factory=$factory flag=1
  end
  * if resources are managed call plugin.LI.FDN.Supply.Factory: factory
  if $settings.array[1]
   * check.time and full.time is used to see if the factory is stalled
   $check.time = $factory -> get remaining production cycle time
   $full.time = $factory -> get production cycle time: account for secondary resources=[FALSE]
   * debug
   if $al.Settings[7]
    $txt.debug = ' Check time = ' + $check.time + ' Full Time = ' + $full.time
    gosub Debug.Sub:
   end
   * v9.01: flags instead of else - stalled, or less than 45 seconds of production left
   $call.resource = [FALSE]
   if $check.time == $full.time
    $call.resource = [TRUE]
    if $al.Settings[7]
     $txt.debug = ' Prdouction stalled - Supply Resource called'
     gosub Debug.Sub:
    end
   end
   if not $call.resource
    if $check.time < 45
     $call.resource = [TRUE]
     if $al.Settings[7]
      $txt.debug = ' Prdouction near completion - Supply Resource called'
      gosub Debug.Sub:
     end
    end
   end
   if $call.resource
    * call the supply factory script to supply resources to the factory
    START null -> call script plugin.LI.FDN.Supply.Factory : factory=$factory flag=null
   end
  end
 end
 * v9.01 B13: next index - backwards always steps down (a removal only shifts entries already checked);
 * forwards only steps up when nothing was removed (the next entry has moved into slot $s)
 if $iterate.backwards
  dec $s =
 end
 if not $iterate.backwards
  if not $removed
   inc $s =
  end
 end
 * wait
 = wait randomly from $wait.1 to $wait.2 ms
end

* v9.01 B1: release the run-lock on exit
$al.Settings[12] = null
set global variable: name='al.LI.FDN.event' value=$al.Settings

return null
* -----------------------------------------------------------------------------------

* Debug Massage handling
Debug.Sub:
$d.time = = playing time
$d.time = = format time: $d.time
$txt = $d.time + ';' + $script + ';' + $txt.debug
write to log file $DebugID  append=[TRUE]  value=$txt
endsub
* -----------------------------------------------------------------------------------
return null
