* ---------- sub - set a station's EC availability to $target ($stn = the station) ----------
* v9.01 D6 (M-04c): floor 0, raise virtual +10000 over the saved value, maintain = virtual + physical - target (min 1) so exactly $target is available; Pay never drains below it
SetAvail.Sub:
$av.rec = [THIS] -> call script lib.LI.FDN.Ware.Rec : station=$stn ware=SS_WARE_ENERGY create=[FALSE]
if is datatyp[ $av.rec ] == DATATYPE_ARRAY
 * no record for this DC: nothing to set, fall through to endsub; the caller's virtual check will SKIP or FAIL
 $dc.cfg.work = $stn -> get local variable: name='FDN.DC.Cfg'
 if not is datatyp[ $dc.cfg.work ] == DATATYPE_ARRAY
  $dc.cfg.work = = array alloc: size=4
  $c0 = $dc.cfg.work[0]
  if not is datatyp[ $c0 ] == DATATYPE_INT
   $dc.cfg.work[0] = -1
  end
 end
 $dc.cfg.work[1] = 0
 $stn -> set local variable: name='FDN.DC.Cfg' value=$dc.cfg.work
 $av.rec[1] = 0
 if is datatyp[ $sav.dc.v0 ] == DATATYPE_INT
  $av.newvirt = $sav.dc.v0 + 10000
 else
  $av.newvirt = 10000
 end
 $av.rec[0] = $av.newvirt
 $av.phys = $stn -> get amount of ware SS_WARE_ENERGY in cargo bay
 if not is datatyp[ $av.phys ] == DATATYPE_INT
  $av.phys = 0
 end
 $av.maint = $av.rec[0] + $av.phys - $target
 if not is datatyp[ $av.maint ] == DATATYPE_INT
  $av.maint = 1
 end
 if $av.maint < 1
  $av.maint = 1
 end
 $av.rec[2] = $av.maint
end
endsub

* ---------- sub - run one Cost call and compare it with $exp ----------
* inputs: $ware $units $jumps $dcx $bb $exp $id (the caller also sets [33] before the gosub)
CostChk.Sub:
$cost.got = [THIS] -> call script lib.LI.FDN.Cost : ware=$ware units=$units jumps=$jumps dc=$dcx backbone=$bb
if not is datatyp[ $cost.got ] == DATATYPE_INT
 * the Cost lib must return an INT; anything else (null / error) fails this check rather than comparing wrong
 $ok = [FALSE]
 $det = 'Cost returned non-INT'
else
 $ok = [FALSE]
 if $cost.got == $exp
  $ok = [TRUE]
 end
 $det = 'got=' + $cost.got + ' expected=' + $exp
end
gosub Result.Sub:
endsub

* sub - recompute the worked-set cost from the lib.LI.FDN.Cost constants and compare against $r.cur / $exp.a.
* Contract (set by the caller before gosub):
*   $exp.wkware  = ware used for this call (the SS_WARE_* constant)
*   $vol.wk      = INT volume of that ware (>0, already verified by the caller)
*   $r.cur       = live result from lib.LI.FDN.Cost
*   $exp.a       = hand-written expected value (for logging; if it disagrees with the recompute we SKIP as data drift)
*   $exp.wkid    = id string for Result.Sub
*   $exp.wkdcsn  = paying DC station or null
*   $exp.wkjumps = INT jumps passed in
*   $exp.wkbb    = [TRUE] backbone, anything else station rates
* No returns: volume not INT or < 1 -> log SKIP, no result; live != recompute -> FAIL via Result.Sub;
* live == recompute but expected differs -> log SKIP "data differs", no result; all agree -> PASS via Result.Sub.
Worked.Sub:
$wk.u = 0
if is datatyp[ $r.cur ] == DATATYPE_INT
 * the live result must be INT for any of this to be meaningful; otherwise SKIP data differs
 if not is datatyp[ $vol.wk ] == DATATYPE_INT
  * caller should have guarded, but stay safe
 end
end
* T per-mille from maintype (mirror lib.LI.FDN.Cost L132-152)

if is datatyp[ $mt.wk ] == DATATYPE_INT
 if not $mt.wk == [SSTYPE_LASER]
  if not $mt.wk == [SSTYPE_SHIELD]
   if not $mt.wk == [SSTYPE_MISSILE]
    if not $mt.wk == [SSTYPE_W_TECH]
     * base group stays 1000
    else
     $wk.t = 1500
    end
   else
    $wk.t = 2000
   end
  else
   $wk.t = 2000
  end
 else
  $wk.t = 2000
 end
end
* V per-mille from average price (mirror L154-167)
$wk.v = 1000
$ap.wk = = get average price of ware $exp.wkware
if is datatyp[ $ap.wk ] == DATATYPE_INT
 if not $ap.wk < 1000
  $wk.v = 1250
 end
 if not $ap.wk < 10000
  $wk.v = 1500
 end
 if not $ap.wk < 100000
  $wk.v = 2000
 end
end
* H per-mille from the paying DC subtype (mirror L169-184); a non-station dc uses the equipment value 1000
$wk.h = 1000
if is datatyp[ $exp.wkdcsn ] == DATATYPE_STATION
 $wk.h = 1200
 $sub.wk = $exp.wkdcsn -> get subtype
 if is datatyp[ $sub.wk ] == DATATYPE_INT
  if $sub.wk == 62
   $wk.h = 800
  end
  if $sub.wk == 6
   $wk.h = 1000
  end
 end

* backbone or station rates (mirror L53-56)

 $wk.J = 2
end
* units and jumps clamped exactly as lib.LI.FDN.Cost does (L96-127) - the caller passes them via $vol.wk / $exp.wkjumps; units are implicit from $r.cur being non-zero.
$wk.u = $exp.wkunits
if not is datatyp[ $wk.u ] == DATATYPE_INT
 * bad/missing units: cost would be 0, nothing to recompute
 if not is datatyp[ $r.cur ] == DATATYPE_INT
 end
end
$wk.j = $exp.wkjumps
if not is datatyp[ $wk.j ] == DATATYPE_INT
 $wk.j = 0
end
if $wk.j < 0
 $wk.j = 0
end
if $wk.j > 50
 $wk.j = 50

* the full integer formula: cost = ceil( U * vol * (B + J*j) * T * V * H / 10^6 ) when units > 0, else 0. No return is used; a bad volume sets $wk.skip so we log SKIP and give no result.
$wk.calc = 0
$wk.skip = [FALSE]
if is datatyp[ $wk.u ] == DATATYPE_INT

  if not is datatyp[ $vol.wk ] == DATATYPE_INT
   * no usable volume: cannot recompute - SKIP, no result
   $txt = 'TEST Cost: WORKED SKIP no INT volume for that ware'
   gosub Log.Sub:
   $wk.skip = [TRUE]
  else

    * volume not > 0 - SKIP, no result
    $txt = 'TEST Cost: WORKED SKIP volume not > 0 (' + $vol.wk + ')'
    gosub Log.Sub:
    $wk.skip = [TRUE]
   else
    * recompute the exact integer cost and compare against the live result and the expected value
    $per.base = $wk.B + $wk.J * $wk.j
    $m.wk = $wk.u * $vol.wk * $per.base
    $f.wk = $wk.t
    gosub Wkmille.Sub:
    $f.wk = $wk.v
    gosub Wkmille.Sub:
    $f.wk = $wk.h
    gosub Wkmille.Sub:
    $wk.calc = ($m.wk + 999) / 1000
    $det.wk = 'got=' + $r.cur + ' recompute=' + $wk.calc + ' expected=' + $exp.a
    if not is datatyp[ $r.cur ] == DATATYPE_INT
     * no live result to compare - SKIP, no result
     $txt = 'TEST Cost: WORKED SKIP no INT result (' + $det.wk + ')'
     gosub Log.Sub:
     $wk.skip = [TRUE]
    else
     if not $r.cur == $wk.calc
      * live result disagrees with our own recompute - data on this box differs, FAIL
      $ok = [FALSE]
      $id = $exp.wkid
      gosub Result.Sub:
     else

       $txt = 'TEST Cost: WORKED SKIP data differs (no INT expected) (' + $det.wk + ')'
       gosub Log.Sub:
       $wk.skip = [TRUE]
      else
       if not $exp.a == $wk.calc
        * hand-written expected disagrees with the formula - SKIP "data differs", no result
        $txt = 'TEST Cost: WORKED SKIP data differs (' + $det.wk + ')'
        gosub Log.Sub:
        $wk.skip = [TRUE]
       else
        * all three agree - count it as a real PASS
        $ok = [TRUE]
        $id = $exp.wkid
        gosub Result.Sub:
       end
      end
     end
    end
   end
  end
 end
end
