* Script: plugin.LI.FDN.Update.Ware  (v8, engine 50)
* Desc: FDN v9 - Add or remove a ware from a station's stock and its virtual ref
* Arg 0: ware (Value) 'ware'
* Arg 1: amount (Number) 'amount'
* Arg 2: station (Value) 'station'
* Arg 3: action (Value) 'action'

* Author: Logain Abler
* -----------------------------------------------------------------------------------
* This script is called by lib.LI.FDN.Trade, plugin.LI.FDN.Supply.Dock,
* plugin.LI.FDN.Supply.Factory, plugin.LI.FDN.Move.Ware and plugin.LI.FDN.Main.Menu.
* It moves wares between physical cargo bay stock and the station local virtual ref
* FDN.Ware.<ware> (record element [0]). Only Dock class stations keep a record; for
* every other station the amount is clamped to the free/physical stock and no record
* is read or written.

* Expected values:
* - ware
* - amount
* - station
* - action ('create', 'add' or 'remove')

* Returns values:
* - 'add' = amount actually added (physical + virtual); 'remove' = amount actually removed; 0 if amount <= 0, if the ware or station is invalid, or if action is unknown; null for 'create' or when FDN settings are missing

* Version: 8.0
* Date: 26/10/2026
* Tested: TC 26/10/2026
* -----------------------------------------------------------------------------------

* v9.01 Phase 2 changes (P2-12a) - closes D1, D2 (one of nine), D3, B2/B9 (via Ware.Rec), B29, B30:
*   D1  Header switched to lib.LI.FDN.Cfg.Get (the only allowed copy of the settings fetch).
*   D2  Debug.Sub deleted; every debug line goes through lib.LI.FDN.Log with $s.name as caller.
*   D3  The ware record is read and created via lib.LI.FDN.Ware.Rec - the text 4000 sprintf,
*       Add.Ware.Sub's array build and the inline settings [29] registry block are gone.
*   B2/B9  Record get-or-create and the station registry add are owned by Ware.Rec (create path).
*   B30  A newly created record is used directly from the Ware.Rec create return - no stale null read.
* -----------------------------------------------------------------------------------

* v9.01 D1: standard header - lib.LI.FDN.Cfg.Get is the only allowed copy of the settings fetch
$s.name = = get script name
$al.Settings = [THIS] -> call script lib.LI.FDN.Cfg.Get :
if not $al.Settings
 return null
end
$debug.on = $al.Settings[7]
* guard - ware must be set (Ware.Rec would build a record name from it, so check first)
if not $ware
 * v9.01 D2: debug - log the guard failure (no variables: station and action may be null here)
 if $debug.on
  $txt.log = 'Update.Ware failed - no ware given'
  = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
 end
 * v9.01: documented return values - null for create, 0 for add/remove/unknown
 if $action == 'create'
  return null
 end
 return 0
end

* guard - station datatype BEFORE any -> call (a number or string would throw a script error)
if not is datatyp[ $station ] == DATATYPE_STATION
 * v9.01 D2: debug - log the guard failure (ware is valid here, station is not)
 if $debug.on
  $txt.log = $ware + ' Update.Ware failed - station is not a station'
  = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
 end
 * v9.01: documented return values - null for create, 0 for add/remove/unknown
 if $action == 'create'
  return null
 end
 return 0
end

* guard - station must still exist
$exists = $station  ->  exists
if not $exists
 * v9.01 D2: debug - log the guard failure (the station is gone, so no station in the text)
 if $debug.on
  $txt.log = $ware + ' Update.Ware failed - station does not exist'
  = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
 end
 * v9.01: documented return values - null for create, 0 for add/remove/unknown
 if $action == 'create'
  return null
 end
 return 0
end

* guard - action must be exactly create, add or remove (an unknown action used to run the remove branch)
$action.ok = [FALSE]
if $action == 'create'
 $action.ok = [TRUE]
end
if $action == 'add'
 $action.ok = [TRUE]
end
if $action == 'remove'
 $action.ok = [TRUE]
end
if not $action.ok
 * v9.01 D2: debug - log the guard failure
 if $debug.on
  $txt.log = $station + ' Update.Ware failed - unknown action, ware ' + $ware
  = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
 end
 return 0
end

* the create action only makes sure a record exists (Ware.Rec never overwrites an existing one)
if $action == 'create'
 $ware.array = [THIS] -> call script lib.LI.FDN.Ware.Rec : station=$station ware=$ware create=[FALSE]
 if not $ware.array
  $ware.array = [THIS] -> call script lib.LI.FDN.Ware.Rec : station=$station ware=$ware create=[TRUE]
  if $ware.array
   gosub Show.Ware.Sub:
   if $debug.on
    $txt.log = $station + ' Record created for ' + $ware
    = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
   end
  end
 end
 return null
end

* nothing to move unless the amount is above 0
if not $amount > 0
 return 0
end

* only a Dock class station keeps a virtual stock record
$virtual.stock = 0
$ware.array = null
$is.dock = $station  ->  is of class Dock 2007
if $is.dock
 $ware.array = [THIS] -> call script lib.LI.FDN.Ware.Rec : station=$station ware=$ware create=[FALSE]
 if not $ware.array
  $ware.array = [THIS] -> call script lib.LI.FDN.Ware.Rec : station=$station ware=$ware create=[TRUE]
  if $ware.array
   gosub Show.Ware.Sub:
   if $debug.on
    $txt.log = $station + ' Record created for ' + $ware
    = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
   end
  end
 end
 if $ware.array
  $virtual.stock = $ware.array[0]
  if not is datatyp[ $virtual.stock ] == DATATYPE_INT
   $virtual.stock = 0
  end
  * write-back name: Ware.Rec reads and creates but cannot store a changed record, so keep this one sprintf (text 4000)
  $PageID = $al.Settings[1]
  $pointer = = sprintf: pageid=$PageID textid=4000, $ware, null, null, null, null
 end
end

$physical.stock = $station  ->  get amount of ware $ware in cargo bay
$physical.free = $station  ->  get free amount of ware $ware in cargo bay

* add: with a record (a Dock) what fits goes into the cargo bay first and the rest into the virtual stock; without one the amount is clamped to the free space
if $action == 'add'
 if $ware.array
  if not $physical.free > $amount
   $to.virtual = $amount - $physical.free
   $virtual.stock = $virtual.stock + $to.virtual
   $ware.array[0] = $virtual.stock
   $station  ->  set local variable: name=$pointer value=$ware.array
    = $station  ->  add $physical.free units of $ware
   if $debug.on
    $txt.log = $station + ' Add ' + $amount + ' of ' + $ware + ', physical ' + $physical.free + ', virtual ' + $to.virtual
    = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
   end
   return $amount
  end
  = $station  ->  add $amount units of $ware
  if $debug.on
   $txt.log = $station + ' Add ' + $amount + ' of ' + $ware + ', physical ' + $amount + ', virtual 0'
   = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
  end
  return $amount
 end
 $to.add = $amount
 if not $physical.free > $amount
  $to.add = $physical.free
 end
  = $station  ->  add $to.add units of $ware
 if $debug.on
  $txt.log = $station + ' Add ' + $to.add + ' of ' + $ware + ' (requested ' + $amount + ')'
  = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
 end
 return $to.add
end

* remove: from the virtual stock first, then from the cargo bay (only 'remove' can reach here, the action was checked above)
$return.remove = 0
$from.virtual = 0
$from.physical = 0
if $amount < $virtual.stock
 $from.virtual = $amount
 $virtual.stock = $virtual.stock - $amount
 if $ware.array
  $ware.array[0] = $virtual.stock
 end
 $return.remove = $amount
else
 $from.virtual = $virtual.stock
 $from.physical = $amount - $virtual.stock
 if $physical.stock < $from.physical
  $from.physical = $physical.stock
 end
 $return.remove = $from.virtual + $from.physical
 if $ware.array
  $ware.array[0] = 0
 end
 $remove = - $from.physical
  = $station  ->  add $remove units of $ware
end

* store the changed record (only a Dock with a record has one)
if $ware.array
 $station  ->  set local variable: name=$pointer value=$ware.array
end
if $debug.on
 $txt.log = $station + ' Remove ' + $return.remove + ' of ' + $ware + ', virtual ' + $from.virtual + ', physical ' + $from.physical + ' (requested ' + $amount + ')'
 = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
end
return $return.remove

return null

Show.Ware.Sub:
* a new record was just made but the ware is not in the cargo bay yet: add 1 and remove 1 so the ware shows on the dock
$physical.check = $station  ->  get amount of ware $ware in cargo bay
if not $physical.check > 0
  = $station  ->  add 1 units of $ware
  = $station  ->  add -1 units of $ware
end
endsub
