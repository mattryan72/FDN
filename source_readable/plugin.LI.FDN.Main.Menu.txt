390    $max.txt = = convert number $max to string

482  $pointer = = sprintf: pageid=$PageID textid=4000, $ware, null, null, null, null
483  $ware.array = $value  ->  get local variable: name=$pointer
484  $virtual = $ware.array[0]
485  
486  if $command == 'to.virtual'
487   $amount = $value  ->  get amount of ware $ware in cargo bay
488   $virtual = $virtual + $amount
489   $ware.array[0] = $virtual
490   $remove = - $amount
491    = $value  ->  add $remove units of $ware
492  else
493   $amount = $value  ->  get free amount of ware $ware in cargo bay
494   if $virtual <= $amount
495     = $value  ->  add $virtual units of $ware
496    $ware.array[0] = 0
497   else
498     = $value  ->  add $amount units of $ware
499    $virtual = $virtual - $amount
500    $ware.array[0] = $virtual
501   end
502  end
503  $value  ->  set local variable: name=$pointer value=$ware.array
504  endsub
