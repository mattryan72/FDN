* Script: plugin.LI.FDN.Main.Menu  (v9.01, engine 50)
* Desc: FDNv9 - Main Menu controller
* Author: Logain Abler
* -----------------------------------------------------------------------------------
* v9.01 P3-05. The controller only: it keeps a local view stack (array of [view, obj, ware],
* top = last), builds the top view, opens the menu once, and hands the return value to
* plugin.LI.FDN.Menu.Command, which returns the NEW stack or null to close. Main.Menu never
* edits the stack and never assigns a view; the old command cascade, the -1 view table,
* the header 'Switch to' row and the show.factory plumbing are gone.
* Views (v8 numbers kept so settings [5] does not change):
*   1 DC (Menu.Station), 2 Stations (Menu.Stations), 4 Factory Details,
*   5 Dock (Menu.Station), 6 Ware Details, 7 Config, 3 DC list (Menu.DC.List, M-06),
*   8 DC running cost (Menu.DC.Cost, M-07g).
* Opening state is [[1, DC, null]], [[3, null, null]] when Dc.List has 2 or more DCs,
* or [[7, null, null]] when no DC is set.
* Back and Escape (-1) are handled by Menu.Command. The footer shows Back (306,
* ['back', null, null, null]) when the stack has more than one entry, and always a last
* 'Close menu' row (307, ['close', null, null, null]) that closes the whole menu (M-06M).
* B8/P3-09: the controller bumps token [20] on every open, and START's the single updater
* plugin.LI.FDN.Menu.Refresh (view, obj, token) only when it ENTERS view 1 or 5 or a different station,
* never on a redraw. Refresh exits by itself for a static view, a newer token or another view.
* Builder contracts (P3-06, P3-07):
*   plugin.LI.FDN.Menu.Station  : menu, view (1 or 5), obj   -> the menu
*   plugin.LI.FDN.Menu.Stations : menu, obj                  -> the menu
*   plugin.LI.FDN.Menu.DC.Cost  : menu, obj (the DC)         -> the menu (M-07g)
* Existing builders keep their arguments:
*   Menu.Factory.Details (menu, value), Menu.Ware.Details (menu, value, ware), Config.Menu (menu)
* Needs texts 306 (Back) and 307 (Close menu) in the text file, plus 302-305 for the titles.
* Menu is called from plugin manager.

* Returns null

* To Do - install together with plugin.LI.FDN.Menu.Refresh (P3-09); the texts 302-307 are in the text file

* Version: 2.4
* Date: 30/10/2026
* Tested: not tested (M-06C, M-06G, M-07g)
* -----------------------------------------------------------------------------------
 * v9.01 D6 (M-06M): footer always ends with 'Close menu' (307, ['close']) that exits all menus; Back (306) only when there is a view to go back to
 * v9.01 D6 (M-06G): view 3 (DC list) opens in the wide info window like views 1, 2 and 5
 * v9.01 D6 (M-06C): view 3 DC list (opens there with 2+ DCs), title 353; scan on menu open ([16] older than 60 s -> null, Supply.Start rescans next tick, never a Scan START here); text 392 note on views 2/3 while a rescan is pending
 * v9.01 D6 (M-07g): view 8 DC running cost (Menu.DC.Cost), title 381, wide info window
 * -----------------------------------------------------------------------------------

$s.name = = get script name
$al.Settings = [THIS] -> call script lib.LI.FDN.Cfg.Get :
if not $al.Settings
 return null
end
$PageID = $al.Settings[1]

* B8: bump the token on every menu open so any _D updater left over from a previous session exits at once
$menu.token = $al.Settings[20]
if not is datatyp[ $menu.token ] == DATATYPE_INT
 $menu.token = 0
end
$menu.token = $menu.token + 1
$al.Settings[20] = $menu.token

* v9.01 D6 (M-06C): scan on menu open. Never call or START Scan here (Scan has no lock and
* Supply.Start owns it): when the last scan [16] is 60 s or older, clear [16] so the next
* Supply.Start tick (25 s timer) STARTs Scan before its run-lock. Not an INT = already due.
$scan.stale = 60
$scan.time = $al.Settings[16]
if is datatyp[ $scan.time ] == DATATYPE_INT
 $now = = playing time
 $age = $now - $scan.time
 if $age >= $scan.stale
  $al.Settings[16] = null
   = [THIS] -> call script lib.LI.FDN.Log : text='Menu open: rescan requested' caller=$s.name
 end
end
set global variable: name='al.LI.FDN.event' value=$al.Settings

* opening stack
$dc = $al.Settings[2]
$entry = array alloc: size=3
if $dc
 $entry[0] = 1
 $entry[1] = $dc
 * v9.01 D6 (M-06C): two or more DCs -> open on the DC list (view 3, Q6.5)
 $dc.list = [THIS] -> call script lib.LI.FDN.Dc.List :
 if is datatyp[ $dc.list ] == DATATYPE_ARRAY
  $n.dc = size of array $dc.list
  if $n.dc >= 2
   $entry[0] = 3
   $entry[1] = null
  end
 end
else
 $entry[0] = 7
end
$stack = array alloc: size=1
$stack[0] = $entry
$updater.view = null
$updater.obj = null
= [THIS] -> call script lib.LI.FDN.Log : text='Menu opened' caller=$s.name

while [TRUE]
 * settings are read again every pass: master.reset can replace the array
 $al.Settings = [THIS] -> call script lib.LI.FDN.Cfg.Get :
 if not $al.Settings
  return null
 end
 $dc = $al.Settings[2]

 * top of the stack
 $size = size of array $stack
 $idx = $size - 1
 $top = $stack[$idx]
 $view = $top[0]
 $obj = $top[1]
 $ware = $top[2]

 * no DC set: the Config view is the only view
 if not $dc
  if not $view == 7
   $entry = array alloc: size=3
   $entry[0] = 7
   $stack = array alloc: size=1
   $stack[0] = $entry
   $view = 7
   $obj = null
   $ware = null
  end
 end

 gosub Updater.Sub:

 $txt = = read text: page=$PageID id=102
 $menu = = create custom menu array: heading=$txt
 gosub Scan.Note.Sub:
 gosub Build.View.Sub:

 $al.Settings[5] = $view
 set global variable: name='al.LI.FDN.event' value=$al.Settings
 add section to custom menu: $menu

 gosub Footer.Sub:

 $new.stack = [THIS] -> call script plugin.LI.FDN.Menu.Command : stack=$stack return=$return
 $close = [TRUE]
 if is datatyp[ $new.stack ] == DATATYPE_ARRAY
  $new.size = size of array $new.stack
  if $new.size > 0
   $close = [FALSE]
   $stack = $new.stack
  end
 end
 if $close == [TRUE]
  = [THIS] -> call script lib.LI.FDN.Log : text='Menu closed' caller=$s.name
  $al.Settings[5] = null
  set global variable: name='al.LI.FDN.event' value=$al.Settings
  START  = null -> call script plugin.LI.FDN.Cleanup :
  return null
 end
end

return null


Updater.Sub:
* B8: START an updater only when entering view 1 or 5 or a different station, bump the token first so an older updater exits
$enter = [FALSE]
if $view == 1
 $enter = [TRUE]
else if $view == 5
 $enter = [TRUE]
end
if not $enter == [TRUE]
 $updater.view = null
 $updater.obj = null
 endsub
end
$same = [FALSE]
if $updater.view == $view
 if $updater.obj == $obj
  $same = [TRUE]
 end
end
if $same == [TRUE]
 endsub
end
$menu.token = $al.Settings[20]
if not is datatyp[ $menu.token ] == DATATYPE_INT
 $menu.token = 0
end
$menu.token = $menu.token + 1
$al.Settings[20] = $menu.token
set global variable: name='al.LI.FDN.event' value=$al.Settings
START  = null -> call script plugin.LI.FDN.Menu.Refresh : view=$view obj=$obj token=$menu.token
$updater.view = $view
$updater.obj = $obj
endsub


Build.View.Sub:
* one builder per view; a builder that returns anything but an array leaves the menu as it was
$built = null
if $view == 1
 $built = [THIS] -> call script plugin.LI.FDN.Menu.Station : menu=$menu view=1 obj=$obj
else if $view == 5
 $built = [THIS] -> call script plugin.LI.FDN.Menu.Station : menu=$menu view=5 obj=$obj
else if $view == 2
 $built = [THIS] -> call script plugin.LI.FDN.Menu.Stations : menu=$menu obj=$obj
else if $view == 4
 $built = [THIS] -> call script plugin.LI.FDN.Menu.Factory.Details : menu=$menu value=$obj
else if $view == 6
 $built = [THIS] -> call script plugin.LI.FDN.Menu.Ware.Details : menu=$menu value=$obj ware=$ware
else if $view == 7
 $built = [THIS] -> call script plugin.LI.FDN.Config.Menu : menu=$menu
else if $view == 3
 * v9.01 D6 (M-06C): DC list
 $built = [THIS] -> call script plugin.LI.FDN.Menu.DC.List : menu=$menu
else if $view == 8
 * v9.01 D6 (M-07g): DC running cost
 $built = [THIS] -> call script plugin.LI.FDN.Menu.DC.Cost : menu=$menu obj=$obj
end
if is datatyp[ $built ] == DATATYPE_ARRAY
 $menu = $built
end
endsub


Scan.Note.Sub:
* v9.01 D6 (M-06C): views 2 and 3 show text 392 at the top while a rescan is pending ([16] not an INT)
$show.note = [FALSE]
if $view == 2
 $show.note = [TRUE]
else if $view == 3
 $show.note = [TRUE]
end
if $show.note == [TRUE]
 $scan.time = $al.Settings[16]
 if not is datatyp[ $scan.time ] == DATATYPE_INT
  $txt = = read text: page=$PageID id=392
  add custom menu info line to array $menu: text=$txt
 end
end
endsub


Footer.Sub:
* Back row (306, ['back']) when there is a view to go back to, then always Close menu
* (307, ['close']), which Menu.Command answers with null, so the whole menu closes
$size = size of array $stack
if $size > 1
 $txt = = read text: page=$PageID id=306
 $back = array alloc: size=4
 $back[0] = 'back'
 = [THIS] -> call script lib.LI.FDN.Menu.Row : menu=$menu label=$txt value=null retval=$back
end
$txt = = read text: page=$PageID id=307
$close = array alloc: size=4
$close[0] = 'close'
= [THIS] -> call script lib.LI.FDN.Menu.Row : menu=$menu label=$txt value=null retval=$close

* window title per view
if $view == 1
 $txt = = read text: page=$PageID id=302
else if $view == 2
 $txt = = read text: page=$PageID id=303
else if $view == 4
 $txt = = read text: page=$PageID id=305

 $txt = = read text: page=$PageID id=304
else if $view == 7
 $txt = = read text: page=$PageID id=103
else if $view == 3
 $txt = = read text: page=$PageID id=353
else if $view == 8
 $txt = = read text: page=$PageID id=381
else
 $txt = = read text: page=$PageID id=133
end

$open.info = [FALSE]
if $view == 1
 $open.info = [TRUE]
else if $view == 2
 $open.info = [TRUE]
else if $view == 5
 $open.info = [TRUE]
else if $view == 3
 $open.info = [TRUE]
else if $view == 8
 $open.info = [TRUE]
end
if $open.info == [TRUE]
 $return = open custom info menu: title=$txt description=null option array=$menu maxoptions=2
else
 $return = open custom menu: title=$txt description=null option array=$menu
