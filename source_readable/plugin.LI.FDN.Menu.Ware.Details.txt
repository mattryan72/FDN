* Called by plugin.LI.FDN.Main.Menu for view 6 as Menu.Ware.Details : menu=$menu value=$obj ware=$ware.
* Shows the station header (Menu.ObjLine), the physical / virtual stock line, and the
* DC or dock settings rows from Build.DC.Rows.Sub / Build.Dock.Rows.Sub. The action buttons
* (move.ware, to.physical, to.virtual, remove.ware) return ['command', station, ware], which
* plugin.LI.FDN.Menu.Command handles. The record is read via lib.LI.FDN.Ware.Rec with
* create=[FALSE]: opening this view never creates a record.

* Expected values:
* - menu (Value): the custom menu array to append to (changed in place)
* - value (Value): the station, the DC dock or any managed dock
* - ware (Value): the ware the view shows

* Returns values:
* - the menu array (the same array that was passed in); bad input adds nothing

 return $menu

 return $menu

$PageID = $al.Settings[1]
$dc = $al.Settings[2]
* datatype check before any -> call (a number or string would throw a script error)
if not is datatyp[ $value ] == DATATYPE_STATION
 return $menu
end
$exists = $value  ->  exists
if not $exists
 return $menu
end
if not $ware
 return $menu
end
* record via Ware.Rec - create=[FALSE]: this view must never create a record
$rec = [THIS] -> call script lib.LI.FDN.Ware.Rec : station=$value ware=$ware create=[FALSE]
if not is datatyp[ $rec ] == DATATYPE_ARRAY
 return $menu
end
 = [THIS] -> call script lib.LI.FDN.Menu.ObjLine : menu=$menu object=$value

* stock line: physical cargo / virtual stock (B4: local normalise only, no write-back)
$amount = $value  ->  get amount of ware $ware in cargo bay
$virtual = $rec[0]
if not $virtual > 0
 $virtual = 0
end
$amount.txt = = convert number $amount to string
$virtual.txt = = convert number $virtual to string
$virtual.txt = = sprintf: pageid=$PageID textid=9004, $virtual.txt, null, null, null, null
$storage = $amount.txt + '/' + $virtual.txt
$label = $ware + ':'
$retval = null
 = [THIS] -> call script lib.LI.FDN.Menu.Row : menu=$menu label=$label value=$storage retval=$retval

* DC or dock settings rows
$txt = = read text: page=$PageID id=187
add custom menu heading to array $menu: title=$txt
$is.dc = 0
* v9.01 D6 (M-08h): any DC in [30] is a DC here; [2] stays as the fallback (list copied to a local first)
$dc.list = $al.Settings[30]
if is datatyp[ $dc.list ] == DATATYPE_ARRAY
 if find $value in array: $dc.list
  $is.dc = 1
 end
end
if $value == $dc
 $is.dc = 1
end
if $is.dc == 1
 gosub Build.DC.Rows.Sub:
else
 gosub Build.Dock.Rows.Sub:
end

* stock action buttons (single-column, stay direct menu items)
$txt = = read text: page=$PageID id=188
add custom menu heading to array $menu: title=$txt
$stock = $virtual + $amount
if $stock > 0
 $txt = = read text: page=$PageID id=177
 $ret.array = = array alloc: size=3
 $ret.array[0] = 'move.ware'
 $ret.array[1] = $value
 $ret.array[2] = $ware
 add custom menu item to array $menu: text=$txt returnvalue=$ret.array
 add section to custom menu: $menu

 $txt = = read text: page=$PageID id=189
 $ret.array = = array alloc: size=3
 $ret.array[0] = 'to.physical'
 $ret.array[1] = $value
 $ret.array[2] = $ware
 add custom menu item to array $menu: text=$txt returnvalue=$ret.array

 $txt = = read text: page=$PageID id=190
 $ret.array = = array alloc: size=3
 $ret.array[0] = 'to.virtual'
 $ret.array[1] = $value
 $ret.array[2] = $ware
 add custom menu item to array $menu: text=$txt returnvalue=$ret.array
end

* remove button - one block for both DC and dock (the v8 code had it twice)
$can.remove = 0
if $stock == 0
 if $is.dc == 0
  $can.remove = 1
 else
  $can.buy = $rec[3]
  if not $can.buy
   $can.remove = 1
  end
 end
end
if $can.remove == 1
 add section to custom menu: $menu
 $txt = = read text: page=$PageID id=191
 $ret.array = = array alloc: size=3
 $ret.array[0] = 'remove.ware'
 $ret.array[1] = $value
 $ret.array[2] = $ware
 add custom menu item to array $menu: text=$txt returnvalue=$ret.array
end

return $menu

* -----------------------------------------------------------------------------------
* Surplus.Text.Sub - output $surplus.txt (v9.01 B5: Default / On / Off)
* [7] null means never set -> Default; the default comes from the global setting [28],
* which also defaults to 1 when not yet written (texts 321-324 come from the text page)
Surplus.Text.Sub:
$s7 = $rec[7]
if is datatyp[ $s7 ] == DATATYPE_NULL
 $s7 = -1
end
* v9.01 P4-07b: an explicit On or Off uses the coloured texts 148 (green ON) and 149 (red OFF); Default (%s) keeps the plain On, Off or v8 compatible text
if $s7 == 1
 $surplus.txt = = read text: page=$PageID id=148
else
 if $s7 == 0
  $surplus.txt = = read text: page=$PageID id=149
 else
  * Default: the global default surplus state (setting [28], null-safe, v9.01 B5)
  $g = $al.Settings[28]
  if is datatyp[ $g ] == DATATYPE_NULL
   $g = 1
  end
  if $g == 1
   $g.txt = = read text: page=$PageID id=321
  else
   if $g == 0
    $g.txt = = read text: page=$PageID id=322
   else
    * v8 compatible - any other stored value shows the v8 state text
    $g.txt = = read text: page=$PageID id=323
   end
  end
  $surplus.txt = = sprintf: pageid=$PageID textid=324, $g.txt, null, null, null, null
 end
end
endsub
