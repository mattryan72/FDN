* Script: plugin.LI.FDN.Move.Ware  (v3, engine 50)
* Desc: FDN v9 - Handles the user inputs for moving wares
* Arg 0: ware (Value) 'ware'
* Arg 1: source (Value) 'source'

* Author: Logain Abler
* -----------------------------------------------------------------------------------
* this script handles the user inputs for moving wares, from a dock or the DC to any
* player owned ship or station. It is called by plugin.LI.FDN.Main.Menu ('move.ware',
* source = the station whose menu is open) and is interactive: two get user input
* prompts can hold the script while the player thinks, so every object is re-checked
* after the amount prompt before anything moves.

* Expected values:
* - ware
* - source (the station whose menu is open)

* Returns values:
* - N/A - always returns null

* Version: 3.0
* Date: 26/10/2026
* Tested: TC 26/10/2026
* -----------------------------------------------------------------------------------
*
* v9.01 P2-12b changes - closes D1, D2 (one of nine), D3 and the B3/B12 source check; keeps B20:
*   D1  Header switched to lib.LI.FDN.Cfg.Get ($PageID comes from $al.Settings[1]).
*   D2  Debug.Sub deleted; every debug line goes through lib.LI.FDN.Log with $s.name as caller.
*   D3  The source ware record is read via lib.LI.FDN.Ware.Rec (create=[FALSE]) - the text 4000
*       sprintf and the null-array read of [0] are gone; a missing record counts as virtual stock 0.
*   B3/B12  The source is validated with lib.LI.FDN.Is.Valid : obj=$source need.dock=[FALSE] up front,
*           and again after the prompts (with $destination -> exists) before anything moves.
*   B20     A cancelled destination prompt exits before any -> call (kept from P1-13a).
* -----------------------------------------------------------------------------------

* v9.01 D1: standard header - lib.LI.FDN.Cfg.Get is the only allowed copy of the settings fetch
$s.name = = get script name
$al.Settings = [THIS] -> call script lib.LI.FDN.Cfg.Get :
if not $al.Settings
 return null
end
$PageID = $al.Settings[1]
$debug.on = $al.Settings[7]

* v9.01 B3/B12: validate the source before any stock read (Is.Valid returns TRUE, FALSE or null)
$src.valid = [THIS] -> call script lib.LI.FDN.Is.Valid : obj=$source need.dock=[FALSE]
if not $src.valid
 if $debug.on
  $txt.log = $source + ' Move.Ware failed - source is invalid'
  = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
 end
 return null
end

* source stock before the prompts (Source.Stock.Sub reads the record and cargo fresh each time it runs)
gosub Source.Stock.Sub:
if not $stock.available > 0
 if $debug.on
  $txt.log = $source + ' has no free ' + $ware + ' - exit move ware'
  = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
 end
 return null
end

* USER SELECT - player owned ship or station
$txt = = read text: page=$PageID id=184
$destination = null  ->  get user input: type=Var/Ship/Station owned by Player, title=$txt
* v9.01 B20: a cancelled prompt returns null - exit before any -> call
if not $destination
 if $debug.on
  $txt.log = $source + ' Destination cancelled - exit move ware'
  = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
 end
 return null
end
if $debug.on
 $txt.log = $source + ' Selected: ' + $destination
 = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
end

* check that the destination can accept the ware (a Dock has virtual storage, anything else needs free cargo space)
$dest.is.dock = $destination  ->  is of class Dock 2007
if not $dest.is.dock
 $dest.free = $destination  ->  get free amount of ware $ware in cargo bay
 if not $dest.free > 0
  $txt = = sprintf: pageid=$PageID textid=194, $destination, $ware, null, null, null
  display subtitle text: text=$txt duration=2000 ms
  return null
 end
end

* catch if source and destination is the same and exit
if $destination == $source
 if $debug.on
  $txt.log = $source + ' Destination is the same as the source - exit move ware'
  = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
 end
 $txt = = sprintf: pageid=$PageID textid=192, $destination, null, null, null, null
 display subtitle text: text=$txt duration=2000 ms
 return null
end

* USER INPUT - amount of ware to move
$txt = = read text: page=$PageID id=178
$amount = null  ->  get user input: type=Var/Number, title=$txt
if is datatyp[ $amount ] == DATATYPE_NULL
 $txt = = sprintf: pageid=$PageID textid=193, $amount, $ware, null, null, null
 display subtitle text: text=$txt duration=2000 ms
 return null
end
if not $amount > 0
 return null
end

* the prompts can stay open a long time: re-check both objects before anything moves
$src.valid = [THIS] -> call script lib.LI.FDN.Is.Valid : obj=$source need.dock=[FALSE]
$dest.exists = $destination  ->  exists
$both.valid = [FALSE]
if $src.valid
 if $dest.exists
  $both.valid = [TRUE]
 end
end
if not $both.valid
 if $debug.on
  $txt.log = $source + ' Move.Ware stopped - source or destination no longer valid'
  = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
 end
 return null
end

* fresh stock after the prompts, then clamp the amount to it
gosub Source.Stock.Sub:
if not $stock.available > 0
 if $debug.on
  $txt.log = $source + ' has no free ' + $ware + ' any more - exit move ware'
  = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
 end
 return null
end
if $amount > $stock.available
 $amount = $stock.available
end
if $debug.on
 $txt.log = $source + ' Input : ' + $amount + ' of ' + $ware + ' to be moved to ' + $destination
 = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
end

* THE MOVE: add to the destination first, then remove what was really added from the source
* Update.Ware only accepts stations, so a station goes through it and a ship gets a plain cargo bay add limited to its free space
$dest.is.station = [FALSE]
if is datatyp[ $destination ] == DATATYPE_STATION
 $dest.is.station = [TRUE]
end
$amount.added = 0
if $dest.is.station
 $amount.added = [THIS] -> call script plugin.LI.FDN.Update.Ware : ware=$ware amount=$amount station=$destination action='add'
else
 $dest.free = $destination  ->  get free amount of ware $ware in cargo bay
 if $dest.free > 0
  $amount.added = $amount
  if $amount.added > $dest.free
   $amount.added = $dest.free
  end
  = $destination  ->  add $amount.added units of $ware
 end
end
if not $amount.added > 0
 if $debug.on
  $txt.log = $source + ' Move of ' + $amount + ' ' + $ware + ' to ' + $destination + ' - nothing could be added'
  = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
 end
 return null
end
if $amount.added < $amount
 $amount = $amount.added
end
$amount.removed = [THIS] -> call script plugin.LI.FDN.Update.Ware : ware=$ware amount=$amount station=$source action='remove'
if $debug.on
 $txt.log = $source + ' Moved ' + $amount + ' of ' + $ware + ' to ' + $destination + ', added ' + $amount.added + ', removed ' + $amount.removed
 = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
end
* the source stock changed in the last instant: log it, never try to undo
if $amount.removed < $amount
 if $debug.on
  $txt.log = $source + ' Move mismatch - added ' + $amount.added + ' but removed only ' + $amount.removed + ' of ' + $ware
  = [THIS] -> call script lib.LI.FDN.Log : text=$txt.log caller=$s.name
 end
end
return null

Source.Stock.Sub:
* read the source record and cargo fresh: virtual stock (no record or no stock counts as 0) plus physical
$ware.array = [THIS] -> call script lib.LI.FDN.Ware.Rec : station=$source ware=$ware create=[FALSE]
$virtual.available = 0
if $ware.array
 $virtual.available = $ware.array[0]
 if not $virtual.available > 0
  $virtual.available = 0
 end
end
$physical.available = $source  ->  get amount of ware $ware in cargo bay
$stock.available = $virtual.available + $physical.available
endsub
