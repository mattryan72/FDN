* Script: plugin.LI.FDN.Menu.Factory.Details  (v901, engine 50)
* Desc: FDN v9.01 - Factory details view of the menu
* Arg 0: menu (Value) 'menu'
* Arg 1: value (Value) 'value'

* Author: Logain Abler
* -----------------------------------------------------------------------------------
* Called by plugin.LI.FDN.Main.Menu for view 4 as Menu.Factory.Details : menu=$menu value=$obj.
* Shows the factory header (Menu.ObjLine), the products and resources it uses with the
* amount per cycle, and the Products / Resources FDN Managed switches. A switch returns
* ['product', factory] or ['resource', factory], which plugin.LI.FDN.Menu.Command handles.
* The factory settings come from lib.LI.FDN.Fac.Cfg with auto.manage null: opening this
* view never marks the factory as seen, so the auto-manage of the first supply pass still
* applies (v8 behaviour).

* Arguments:
* - menu (Value): the custom menu array to append to (changed in place)
* - value (Value): the factory station

* Returns values:
* - the menu array (the same array that was passed in); a bad station adds nothing

* To Do - N/A

* Version: 1.0
* Date: 27/10/2026
* Tested: FL 27/10/2026 (builder test 6/6, AP branch; TC branch not yet seen)
* -----------------------------------------------------------------------------------
 * v9.01 D6 (P3-08a): hand-built info and toggle rows replaced by Menu.ObjLine and Menu.Toggle
 * -----------------------------------------------------------------------------------

if not is datatyp[ $menu ] == DATATYPE_ARRAY
 return $menu
end
$al.Settings = [THIS] -> call script lib.LI.FDN.Cfg.Get :
if not $al.Settings
 return $menu
end
$PageID = $al.Settings[1]
* datatype check before any -> call (a number or string would throw a script error)
if not is datatyp[ $value ] == DATATYPE_STATION
 return $menu
end
$exists = $value  ->  exists
if not $exists
 return $menu
end
$settings.array = [THIS] -> call script lib.LI.FDN.Fac.Cfg : factory=$value auto.manage=null
if not is datatyp[ $settings.array ] == DATATYPE_ARRAY
 return $menu
end
 = [THIS] -> call script lib.LI.FDN.Menu.ObjLine : menu=$menu object=$value

* split the tradeable wares into products and resources
$product.array = = array alloc: size=0
$resource.array = = array alloc: size=0
$ware.array = $value  ->  get tradeable ware array from station
$s = size of array $ware.array
while $s > 0
 dec $s =
 $ware = $ware.array[$s]
 $is.product = $value  ->  uses ware $ware as product
 if $is.product
  append $ware to array $product.array
 else
  append $ware to array $resource.array
 end
end

* products with the amount per cycle
$txt = = read text: page=$PageID id=137
add custom menu info line to array $menu: text=$txt
$s = size of array $product.array
while $s > 0
 dec $s =
 $ware = $product.array[$s]
 $amount = = $value  ->  get number of products per cycle
 $txt = $ware + ' [' + $amount + ']'
 $temp.array = = array alloc: size=2
 $temp.array[0] = -1
 $temp.array[1] = $txt
 add custom menu info line to array $menu: text=$temp.array
end
add custom menu info line to array $menu: text=' '

* resources with the amount per cycle
$txt = = read text: page=$PageID id=139
add custom menu info line to array $menu: text=$txt
$s = size of array $resource.array
while $s > 0
 dec $s =
 $ware = $resource.array[$s]
 $amount = = $value  ->  get number of resources per cycle for ware $ware
 $txt = $ware + ' [' + $amount + ']'
 $temp.array = = array alloc: size=2
 $temp.array[0] = -1
 $temp.array[1] = $txt
 add custom menu info line to array $menu: text=$temp.array
end

* Products On/Off
$txt = = read text: page=$PageID id=137
add custom menu heading to array $menu: title=$txt
$label = = read text: page=$PageID id=138
$ret.array = = array alloc: size=2
$ret.array[0] = 'product'
$ret.array[1] = $value
$on = $settings.array[0]
 = [THIS] -> call script lib.LI.FDN.Menu.Toggle : menu=$menu label=$label bool=$on retval=$ret.array

* Resource On/Off
$txt = = read text: page=$PageID id=139
add custom menu heading to array $menu: title=$txt
$label = = read text: page=$PageID id=138
$ret.array = = array alloc: size=2
$ret.array[0] = 'resource'
$ret.array[1] = $value
$on = $settings.array[1]
 = [THIS] -> call script lib.LI.FDN.Menu.Toggle : menu=$menu label=$label bool=$on retval=$ret.array

return $menu
