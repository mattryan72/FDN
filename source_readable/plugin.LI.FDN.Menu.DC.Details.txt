036  if $dc
037   * -----------------------------------------------------------------------------------

042   $txt = = read text: page=$PageID id=115
043   add custom menu heading to array $menu: title=$txt
044   * -----------------------------------------------------------------------------------
045   $txt = = read text: page=$PageID id=129
046   add custom menu info line to array $menu: text=$txt
047   * -----------------------------------------------------------------------------------
048   * Calls script to format name
049   $txt = [THIS] -> call script plugin.LI.FDN.Format.Name : object=$dc

051   $temp.array = = array alloc: size=4
052   $temp.array[0] = 1

061   * $sorted.ware.array[x][x] is the sorted Ware Array for the Main Type

067   $pointer = = read text: page=$PageID id=4004

073   * -----------------------------------------------------------------------------------
074

078   * -----------------------------------------------------------------------------------
079   * Start of While Loop with 1 nested While Loop - First loop is Main Type
080   * -----------------------------------------------------------------------------------
081   while $sm > 0
082    dec $sm =
083    * -----------------------------------------------------------------------------------
084    * Fetch main.type and ware.array from sorted.ware.array

086    $ware.array = $sorted.ware.array[$sm][1]
087    * -----------------------------------------------------------------------------------
088    $sw = size of array $ware.array
089    if $sw > 0
090     * -----------------------------------------------------------------------------------
091     * if ware.show.array for the main type is null set to 1 (1 = show 2 = hide)
092     if not $show.type.array[$sm]

095     * -----------------------------------------------------------------------------------
096     * -----------------------------------------------------------------------------------
097     * If show wares Start of While Loop - Second Loop is Wares
098     $check = $show.type.array[$sm]
099     gosub Get.Ware.Header.Sub:
100     if $check == 1
101      * -----------------------------------------------------------------------------------
102      while $sw > 0
103       dec $sw =
104       $ware = $ware.array[$sw]

110       append $summary to array $menu
111       * -----------------------------------------------------------------------------------
112      end
113     end
114    end
115    add section to custom menu: $menu
116   end
117  else
118   * -----------------------------------------------------------------------------------
119   * Show message - No DC
120   $txt = = read text: page=$PageID id=115
121   add custom menu heading to array $menu: title=$txt

129  $return.array = = array alloc: size=2
130  $return.array[0] = 'add.ware'

136  $txt = = read text: page=$PageID id=164
137  add custom menu heading to array $menu: title=$txt
138  $txt = = read text: page=$PageID id=165
139  add custom menu item to array $menu: text=$txt returnvalue='open.config'
140  * -----------------------------------------------------------------------------------
141  * -----------------------------------------------------------------------------------

150  * This Sub is used to create the wares header by main type

152  if $check == 1
153   $temp.array = = array alloc: size=12
154   $temp.array[0] = 1
155   $txt = = read text: page=$PageID id=$main.type
156   $txt = = sprintf: pageid=$PageID textid=18, $txt, $sw, null, null, null
157   $temp.array[1] = $txt

160   $temp.array[3] = $txt

162   $txt = = read text: page=$PageID id=144
163   $temp.array[5] = $txt
164   $temp.array[6] = -290
165   $txt = = read text: page=$PageID id=143
166   $temp.array[7] = $txt
167   $temp.array[8] = -200
168   $txt = = read text: page=$PageID id=145
169   $temp.array[9] = $txt
170   $temp.array[10] = -1
171   $txt = = read text: page=$PageID id=140
172   $temp.array[11] = $txt
173  else
174   $temp.array = = array alloc: size=2
175   $temp.array[0] = 1
176   $txt = = read text: page=$PageID id=$main.type
177   $txt = = sprintf: pageid=$PageID textid=19, $txt, $sw, null, null, null
178   $temp.array[1] = $txt
179  end
180  
181  $return.array = = array alloc: size=2
182  $return.array[0] = 'show.type.dc'
183  $return.array[1] = $sm
184
