* Script: plugin.LI.FDN.Config.Menu  (v3, engine 50)
* Desc: FDNv9 - Config menu (view 7)
* Arg 0: menu (Value) 'menu'

* Author: Logain Abler
* -----------------------------------------------------------------------------------
* Called by plugin.LI.FDN.Main.Menu for view 7. Appends the DC choice, dynamic menu
* switches, global options, debug switch and master reset rows to the passed-in menu.
* Every clickable row returns an array [command, object] handled by
* plugin.LI.FDN.Menu.Command.

* Expected values:
* - menu (Value)

* Returns values:
* - the menu

* To Do - N/A

* Version: 1.0
* Date: 27/10/2026
* Tested: FL 27/10/2026 (test.LI.FDN.ConfigMenu 9/10, see notes/P3-08c.md)
* -----------------------------------------------------------------------------------
 * v9.01 D6 (P3-08c): rows on Menu.Row / Menu.Toggle / Menu.ObjLine; add.dc and change.dc merged into set.dc; B5 surplus.default row
* -----------------------------------------------------------------------------------

if not is datatyp[ $menu ] == DATATYPE_ARRAY
 return $menu
end
$al.Settings = [THIS] -> call script lib.LI.FDN.Cfg.Get :
if not $al.Settings
 return $menu
end
$PageID = $al.Settings[1]
$dc = $al.Settings[2]
$is.dynamic = $al.Settings[6]
if not is datatyp[ $is.dynamic ] == DATATYPE_ARRAY
 $is.dynamic = = array alloc: size=3
end

* valid DC flag: a DC that is unset or no longer a valid dock shows the add list
$has.dc = 0
if is datatyp[ $dc ] == DATATYPE_STATION
 $valid = [THIS] -> call script lib.LI.FDN.Is.Valid : obj=$dc need.dock=[TRUE]
 if $valid == [TRUE]
  $has.dc = 1
 end
end

* DC choice (add / change) or the current DC header
if $has.dc == 0
 * no valid DC yet: the add-DC list under its heading text 123
 $txt = = read text: page=$PageID id=118
 add custom menu info line to array $menu: text=$txt
 $cmd.label = = read text: page=$PageID id=123
 gosub Dock.List.Sub:
else
 * valid DC in place: header with ObjLine, then the change-DC list (heading text 124)
 $txt = = read text: page=$PageID id=129
 add custom menu info line to array $menu: text=$txt
  = [THIS] -> call script lib.LI.FDN.Menu.ObjLine : menu=$menu object=$dc
 $cmd.label = = read text: page=$PageID id=124
 gosub Dock.List.Sub:
end

gosub Dynamic.Rows.Sub:
gosub Global.Rows.Sub:
gosub Debug.Rows.Sub:
gosub Reset.Rows.Sub:

return $menu
* -----------------------------------------------------------------------------------
* SUB Section


Dock.List.Sub:
add custom menu heading to array $menu: title=$cmd.label
$dock.array = = get station array: of race Player class/type=Dock 2007
$s = size of array $dock.array
while $s > 0
 dec $s =
 $dock = $dock.array[$s]
 if $dock
  $valid = [THIS] -> call script lib.LI.FDN.Is.Valid : obj=$dock need.dock=[TRUE]
  if $valid == [TRUE]
   if not $dock == $dc
    $name.txt = [THIS] -> call script plugin.LI.FDN.Format.Name : object=$dock
    if is datatyp[ $name.txt ] == DATATYPE_STRING
     $sector = $dock -> get sector
     $ret.array = = array alloc: size=2
     $ret.array[0] = 'set.dc'
     $ret.array[1] = $dock
      = [THIS] -> call script lib.LI.FDN.Menu.Row : menu=$menu label=$name.txt value=$sector retval=$ret.array
    end
   end
  end
 end
end
endsub


Dynamic.Rows.Sub:
$txt = = read text: page=$PageID id=125
add custom menu heading to array $menu: title=$txt

* dock dynamic DC (switch [0])
$on.off = $is.dynamic[0]
$txt = = read text: page=$PageID id=104
$ret.array = = array alloc: size=1
$ret.array[0] = 'dynamic.dc'
 = [THIS] -> call script lib.LI.FDN.Menu.Toggle : menu=$menu label=$txt bool=$on.off retval=$ret.array

* dock dynamic dock (switch [2])
$on.off = $is.dynamic[2]
$txt = = read text: page=$PageID id=106
$ret.array = = array alloc: size=1
$ret.array[0] = 'dynamic.dock'
 = [THIS] -> call script lib.LI.FDN.Menu.Toggle : menu=$menu label=$txt bool=$on.off retval=$ret.array
endsub


Global.Rows.Sub:
$txt = = read text: page=$PageID id=108
add custom menu heading to array $menu: title=$txt

* auto (global [9])
$on.off = $al.Settings[9]
$txt = = read text: page=$PageID id=109
$ret.array = = array alloc: size=1
$ret.array[0] = 'auto'
 = [THIS] -> call script lib.LI.FDN.Menu.Toggle : menu=$menu label=$txt bool=$on.off retval=$ret.array

* surplus default (global [28]): 1 on, 0 off, 2 v8 compatible; null counts as 1
$g = $al.Settings[28]
if is datatyp[ $g ] == DATATYPE_NULL
 $g = 1
end
if $g == 1
 $g.txt = = read text: page=$PageID id=321
else
 if $g == 0
  $g.txt = = read text: page=$PageID id=322
 else
  $g.txt = = read text: page=$PageID id=323
 end
end
$txt = = read text: page=$PageID id=320
$ret.array = = array alloc: size=1
$ret.array[0] = 'surplus.default'
 = [THIS] -> call script lib.LI.FDN.Menu.Row : menu=$menu label=$txt value=$g.txt retval=$ret.array
$txt = = read text: page=$PageID id=325
add custom menu info line to array $menu: text=$txt
endsub


Debug.Rows.Sub:
$txt = = read text: page=$PageID id=126
add custom menu heading to array $menu: title=$txt

* debug (global [7])
$on.off = $al.Settings[7]
$txt = = read text: page=$PageID id=107
$ret.array = = array alloc: size=1
$ret.array[0] = 'debug'
 = [THIS] -> call script lib.LI.FDN.Menu.Toggle : menu=$menu label=$txt bool=$on.off retval=$ret.array
endsub


Reset.Rows.Sub:
$txt = = read text: page=$PageID id=127
add custom menu heading to array $menu: title=$txt
$txt = = read text: page=$PageID id=128
add custom menu item to array $menu: text=$txt returnvalue='master.reset'
endsub
